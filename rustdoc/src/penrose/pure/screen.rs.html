7/// A wrapper around a single [Workspace] that includes the physical screen
8/// size as a [Rect].
9#[derive(Default, Debug, Clone)]
10pub struct Screen<C> {
11    pub(crate) index: usize,
12    /// The [Workspace] current visible on this screen
13    pub workspace: Workspace<C>,
14    pub(crate) r: Rect,
15}
16
17impl<C: fmt::Display> fmt::Display for Screen<C> {
18    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
19        write!(
20            f,
21            "Screen({}: {:?}):\n  - workspace: {}",
22            self.index, self.r, self.workspace
23        )
24    }
25}

27impl<C> Screen<C> {
28    /// The index of this screen.
29    ///
30    /// Indices are assigned from left to right based on the absolute position of

33        self.index
34    }

36    /// The physical screen size of this [Screen] as a [Rect].

42impl<C> Screen<C>
43where
44    C: Clone + Eq + Hash,
45{
46    /// Produce a summary of the clients contained within this screen
47    pub fn screen_clients(&self, floating: &HashMap<C, RelativeRect>) -> ScreenClients<C> {
48        ScreenClients {
49            floating: self
50                .workspace
51                .clients()
52                .flat_map(|c| floating.get(c).map(|r| (c.clone(), *r)))
53                .collect(),
54            tiling: self
55                .workspace
56                .stack
57                .as_ref()
58                .and_then(|st| st.from_filtered(|c| !floating.contains_key(c))),
59            tag: self.workspace.tag.clone(),
60            r_s: self.r,
61        }
62    }
63}
