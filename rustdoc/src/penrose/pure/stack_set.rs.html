43    pub fn try_new<I, J, T>(layouts: LayoutStack, ws_tags: I, screen_details: J) -> Result<Self>
44    where
45        T: Into<String>,
46        I: IntoIterator<Item = T>,
47        J: IntoIterator<Item = Rect>,
48    {
49        let workspaces: Vec<Workspace<C>> = ws_tags
50            .into_iter()
51            .enumerate()
52            .map(|(i, tag)| Workspace::new(i, tag, layouts.clone(), None))
53            .collect();

56
57        Self::try_new_concrete(workspaces, screen_details, HashMap::new())
58    }

60    pub(crate) fn try_new_concrete(
61        mut workspaces: Vec<Workspace<C>>,
62        screen_details: Vec<Rect>,
63        floating: HashMap<C, RelativeRect>,
64    ) -> Result<Self> {
65        check_workspace_invariants(&workspaces)?;
66
67        match (workspaces.len(), screen_details.len()) {
68            (_, 0) => return Err(Error::NoScreens),
69            (n_ws, n_screens) if n_ws < n_screens => {
70                return Err(Error::InsufficientWorkspaces { n_ws, n_screens })
71            }
72            _ => (),
73        }
74
75        let hidden: VecDeque<Workspace<C>> = workspaces
76            .split_off(screen_details.len())
77            .into_iter()
78            .collect();
79
80        let screens =
81            Stack::from_iter_unchecked(workspaces.into_iter().zip(screen_details).enumerate().map(
82                |(index, (workspace, r))| Screen {
83                    workspace,
84                    index,
85                    r,
86                },
87            ));
88
89        let previous_tag = screens.focus.workspace.tag.clone();
90
91        Ok(Self {
92            screens,
93            hidden,
94            floating,
95            previous_tag,
96            invisible_tags: vec![],
97            killed_clients: vec![],
98        })
99    }

125    /// Set focus to the [Workspace] with the specified tag.
126    ///
127    /// If there is no matching workspace or the provided tag for an invisible
128    /// workspace then the [StackSet] is unmodified. (See
129    /// [add_invisible_workspace](Self::add_invisible_workspace) for details).
130    /// If the [Workspace] is currently visible then focus moves to the screen
131    /// containing that workspace, otherwise the workspace replaces whatever
132    /// was on the active screen.
133    ///
134    /// If you always want to focus the given tag on the active screen, see
135    /// [StackSet::pull_tag_to_screen] instead.
136    pub fn focus_tag(&mut self, tag: impl AsRef<str>) {
137        let tag = tag.as_ref();
138
139        if self.screens.focus.workspace.tag == tag || self.invisible_tags.iter().any(|t| t == tag) {
140            return;
141        }
142
143        // If the tag is visible on another screen, focus moves to that screen
144        if !self.try_cycle_screen_to_tag(tag) {
145            // If the tag is hidden then it gets moved to the current screen
146            self.try_swap_on_screen_workspace_with_hidden(tag);
147        }
148
149        // If nothing matched by this point then the requested tag is unknown
150        // so there is nothing for us to do
151    }
152
153    fn try_cycle_screen_to_tag(&mut self, tag: &str) -> bool {
154        let current_tag = self.screens.focus.workspace.tag.clone();
155
156        loop {
157            self.screens.focus_down();
158            match &self.screens.focus.workspace.tag {
159                // we've found and focused the tag
160                t if t == tag => {
161                    self.update_previous_tag(current_tag);
162                    return true;
163                }
164
165                // we've looped so this tag isn't visible
166                t if t == &current_tag => return false,
167
168                // try the next tag
169                _ => (),
170            }
171        }
172    }

174    fn try_swap_on_screen_workspace_with_hidden(&mut self, tag: &str) {
175        if let Some(mut w) = pop_where!(self, hidden, |w: &Workspace<C>| w.tag == tag) {
176            self.update_previous_tag(self.screens.focus.workspace.tag.clone());
177            swap(&mut w, &mut self.screens.focus.workspace);
178            self.hidden.push_back(w);
179        }
180    }

182    // true if we swapped otherwise false
183    fn try_swap_focused_workspace_with_tag(&mut self, tag: &str) -> bool {
184        if self.screens.focus.workspace.tag == tag {
185            return false;
186        }
187
188        let p = |s: &&mut Screen<C>| s.workspace.tag == tag;
189
190        let in_up = self.screens.up.iter_mut().find(p);
191        let in_down = self.screens.down.iter_mut().find(p);
192
193        if let Some(s) = in_up.or(in_down) {
194            swap(&mut self.screens.focus.workspace, &mut s.workspace);
195            return true;
196        }
197
198        false
199    }

201    /// Focus the requested tag on the current screen, swapping the current
202    /// tag with it.
203    pub fn pull_tag_to_screen(&mut self, tag: impl AsRef<str>) {
204        let tag = tag.as_ref();

206        if self.screens.focus.workspace.tag == tag {
207            return;
208        }
209

211            self.try_swap_on_screen_workspace_with_hidden(tag);

220    /// Focus the given client and set its [Workspace] as current (see
221    /// focus_tag).
222    ///
223    /// If the client is unknown or on an invisible tag then then this is a no-op.
224    pub fn focus_client(&mut self, client: &C) {
225        if self.current_client() == Some(client) {
226            return; // already focused
227        }
228
229        let tag = match self.tag_for_client(client) {
230            Some(tag) if self.invisible_tags.iter().any(|t| t == tag) => return,
231            None => return, // unknown client
232            Some(tag) => tag.to_string(),
233        };
234
235        self.focus_tag(&tag);
236        debug_assert_eq!(
237            self.current_tag(),
238            tag,
239            "attempt to focus unknown or invisible tag for client"
240        );
241
242        while self.current_client() != Some(client) {
243            self.focus_up()
244        }
245    }

430    fn focus_adjacent_workspace(&mut self, tags: Vec<String>) {
431        let cur_tag = self.current_tag();
432        let mut it = tags.iter().skip_while(|t| *t != cur_tag);
433        // We should never be without at least our focused tag as a Screen requires a Workspace to
434        // be present on it, but handling things this way means we avoid having any unwrapping or
435        // raw indexing into vecs.
436        if let Some(new_tag) = it.nth(1).or(tags.first()) {
437            self.pull_tag_to_screen(new_tag)
438        }
439    }

538    /// Attempt to set a new tag for an existing [Workspace].
539    ///
540    /// This will error if the new tag collides with an existing one or if the
541    /// old tag is unknown.
542    pub fn try_rename_workspace(
543        &mut self,
544        old_tag: impl AsRef<str>,
545        new_tag: impl Into<String>,
546    ) -> Result<()> {
547        let new_tag = new_tag.into();
548        if old_tag.as_ref() == new_tag {
549            return Ok(());
550        }
551
552        if self.contains_tag(&new_tag) {
553            return Err(Error::NonUniqueTags {
554                tags: vec![new_tag],
555            });
556        }
557
558        match self.workspace_mut(old_tag.as_ref()) {
559            Some(ws) => {
560                ws.tag = new_tag;
561                Ok(())
562            }
563            None => Err(Error::UnknownWorkspace(old_tag.as_ref().to_string())),
564        }
565    }

567    /// Add a new [Workspace] to this [StackSet].
568    ///
569    /// The id assigned to this workspace will be max(workspace ids) + 1.
570    ///
571    /// # Errors
572    /// This function will error with `NonUniqueTags` if the given tag is already present.
573    pub fn add_workspace<T>(&mut self, tag: T, layouts: LayoutStack) -> Result<()>
574    where
575        T: Into<String>,
576    {
577        let tag = tag.into();
578        if self.contains_tag(&tag) {
579            return Err(Error::NonUniqueTags { tags: vec![tag] });
580        }
581
582        let id = self
583            .workspaces()
584            .map(|w| w.id)
585            .max()
586            .expect("at least one workspace")
587            + 1;
588        let ws = Workspace::new(id, tag, layouts, None);
589        self.hidden.push_front(ws);
590
591        Ok(())
592    }

594    /// Attempt to remove a [Workspace] from this [StackSet] by tag.
595    ///
596    /// Removing a workspace in this way will close all clients present on the workspace.
597    ///
598    /// If the workspace tag is unknown or if removal would result in insufficient workspaces
599    /// for the current number of screens then this method will return None, otherwise it will
600    /// return the Workspace that was removed.
601    pub fn remove_workspace(&mut self, tag: impl AsRef<str>) -> Option<Workspace<C>> {
602        let tag = tag.as_ref();
603
604        for s in self.screens.iter_mut() {
605            if s.workspace.tag != tag {
606                continue;
607            }
608
609            // If we don't have a replacement in hidden we cant remove
610            let mut ws = self.hidden.pop_front()?;
611            swap(&mut ws, &mut s.workspace);
612
613            if self.previous_tag == tag {
614                self.previous_tag = s.workspace.tag.clone();
615            }
616
617            return Some(ws);
618        }
619
620        let opt = pop_where!(self, hidden, |w: &Workspace<C>| w.tag == tag);
621
622        if self.previous_tag == tag {
623            self.previous_tag = self.screens.focus.workspace.tag.clone();
624        }
625
626        opt
627    }

907    pub(crate) fn update_screens(&mut self, rects: Vec<Rect>) -> Result<()> {
908        let n_old = self.screens.len();
909        let n_new = rects.len();
910
911        if n_new == 0 {
912            return Err(Error::NoScreens);
913        }
914
915        match n_new.cmp(&n_old) {
916            // Just a change in dimensions
917            Ordering::Equal => (),
918
919            // We have more screens now: pull in hidden workspaces to fill them
920            // If we run out of workspaces we backfill using generated defaults
921            Ordering::Greater => {
922                let padding = self.take_from_hidden(n_new - n_old);
923                for (n, w) in padding.into_iter().enumerate() {
924                    self.screens.insert_at(
925                        Position::Tail,
926                        Screen {
927                            workspace: w,
928                            index: n_old + n,
929                            r: Rect::default(),
930                        },
931                    );
932                }
933            }
934
935            // We have fewer screens now: focus moves to the first screen and
936            // we drop from the back of the stack
937            Ordering::Less => {
938                let mut raw = take(&mut self.screens).flatten();
939                let removed = raw.split_off(n_new);
940                self.hidden.extend(removed.into_iter().map(|s| s.workspace));
941                self.screens = Stack::from_iter_unchecked(raw);
942            }
943        }
944
945        // self.screens.len() is now correct so update the screen dimensions
946        for (s, r) in self.screens.iter_mut().zip(rects) {
947            s.r = r;
948        }
949
950        Ok(())
951    }
952
953    // This is a little fiddly...
954    // Rather than hard erroring if we end up with new screens being detected that
955    // push us over the number of available workspaces, we pad the workspace set
956    // with ones we generate with default values. In doing this we need to make sure
957    // that any _invisible_ workspaces are kept to one side so that they do not end
958    // up focused on a screen by mistake.
959    fn take_from_hidden(&mut self, n: usize) -> Vec<Workspace<Xid>> {
960        let next_id = self.workspaces().map(|w| w.id).max().unwrap_or(0) + 1;
961        let mut tmp = Vec::with_capacity(self.hidden.len());
962        let mut hidden = VecDeque::new();
963
964        // Filter out any hidden tags first
965        for w in take(&mut self.hidden) {
966            if self.invisible_tags.contains(&w.tag) {
967                hidden.push_front(w);
968            } else {
969                tmp.push(w);
970            }
971        }
972
973        // Sort so that we populate the new screens with workspaces in order of ID.
974        // Without this we are basing things off of the order we ended up with after
975        // whatever workspace focus changes the user has made while we are running.
976        tmp.sort_by_key(|w| w.id);
977
978        // Pad the remaining workspace count with empty default workspaces if we are
979        // below the number we need.
980        if tmp.len() < n {
981            for m in 0..(n - tmp.len()) {
982                tmp.push(Workspace::new_default(next_id + m));
983            }
984        } else {
985            let extra = tmp.split_off(n);
986            hidden.extend(extra);
987        }
988
989        self.hidden = hidden;
990
991        tmp
992    }
