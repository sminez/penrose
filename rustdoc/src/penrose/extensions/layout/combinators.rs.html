1//! Higher order combinators for Layouts that allow for composing their behaviour
2use crate::{
3    core::layout::{Layout, Message},
4    pure::{geometry::Rect, Stack},
5    Xid,
6};
7use std::fmt;
8
9/// Conditionally run one of two layouts based on a predicate function.
10///
11/// This struct implements [Layout] by selecting between the two provided layouts using
12/// a predicate function. By default the left layout will be used, switching to the right
13/// when the predicate returns false. Examples of predicate functions that might be useful are:
14///   - When the screen size being laid out is smaller than a given threshold
15///   - When there are more than a given number of clients that need to be laid out
16///   - Based on the absolute position of the screen being laid out.
17pub struct Conditional {
18    name: String,
19    left: Box<dyn Layout>,
20    right: Box<dyn Layout>,
21    should_use_left: fn(&Stack<Xid>, Rect) -> bool,
22    left_is_active: bool,
23}
24
25impl fmt::Debug for Conditional {
26    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
27        f.debug_struct("Conditional")
28            .field("name", &self.name)
29            .field("left", &self.left.name())
30            .field("right", &self.right.name())
31            .field("left_is_active", &self.left_is_active)
32            .finish()
33    }
34}
35
36impl Conditional {
37    /// Construct a new [Conditional] layout, selecting from one of two layouts based on
38    /// a predicate function.
39    pub fn new<L: Layout + 'static, R: Layout + 'static>(
40        name: impl Into<String>,
41        left: L,
42        right: R,
43        should_use_left: fn(&Stack<Xid>, Rect) -> bool,
44    ) -> Self {
45        Self {
46            name: name.into(),
47            left: Box::new(left),
48            right: Box::new(right),
49            should_use_left,
50            left_is_active: true,
51        }
52    }
53
54    /// Create a new [Conditional] layout as with `new` but returned as a trait
55    /// object ready to be added to your layout stack in config.
56    pub fn boxed<L: Layout + 'static, R: Layout + 'static>(
57        name: impl Into<String>,
58        left: L,
59        right: R,
60        should_use_left: fn(&Stack<Xid>, Rect) -> bool,
61    ) -> Box<dyn Layout> {
62        Box::new(Self::new(name, left, right, should_use_left))
63    }
64}
65
66impl Layout for Conditional {
67    fn name(&self) -> String {
68        self.name.clone()
69    }
70
71    fn boxed_clone(&self) -> Box<dyn Layout> {
72        Box::new(Self {
73            name: self.name.clone(),
74            left: self.left.boxed_clone(),
75            right: self.right.boxed_clone(),
76            should_use_left: self.should_use_left,
77            left_is_active: self.left_is_active,
78        })
79    }
80
81    fn layout(&mut self, s: &Stack<Xid>, r: Rect) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>) {
82        self.left_is_active = (self.should_use_left)(s, r);
83        if self.left_is_active {
84            self.left.layout(s, r)
85        } else {
86            self.right.layout(s, r)
87        }
88    }
89
90    fn handle_message(&mut self, m: &Message) -> Option<Box<dyn Layout>> {
91        if self.left_is_active {
92            self.left.handle_message(m)
93        } else {
94            self.right.handle_message(m)
95        }
96    }
97}
