1//! Startup hooks for direct adding to your penrose config.
2use crate::{
3    Result,
4    core::{State, conn::Conn, hooks::StateHook},
5    util::spawn,
6    x::{Atom, Prop, XConn},
7};
8use std::borrow::Cow;

10/// Spawn a client program on window manager startup
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub struct SpawnOnStartup {
13    prog: Cow<'static, str>,
14}
15
16impl SpawnOnStartup {
17    /// Create a new unboxed startup hook ready for adding to your Config
18    pub fn new(prog: impl Into<Cow<'static, str>>) -> Self {
19        Self { prog: prog.into() }
20    }
21
22    /// Create a new startup hook ready for adding to your Config
23    pub fn boxed<C>(prog: impl Into<Cow<'static, str>>) -> Box<dyn StateHook<C>>
24    where
25        C: Conn,
26    {
27        Box::new(Self::new(prog))
28    }
29}
30
31impl<C> StateHook<C> for SpawnOnStartup
32where
33    C: Conn,
34{
35    fn call(&mut self, _state: &mut State<C>, _: &mut C) -> Result<()> {
36        spawn(self.prog.as_ref())
37    }
38}

40/// Remove _NET_WM_STATE_FULLSCREEN property from existing clients, not present in
41/// [client_set][0], on window manager startup.
42///
43///   [0]: crate::core::State::client_set
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub struct ClearFsPropOnStartup;
46
47impl ClearFsPropOnStartup {
48    /// Create a new startup hook ready for adding to your Config
49    pub fn boxed<X>() -> Box<dyn StateHook<X>>
50    where
51        X: XConn,
52    {
53        Box::new(Self)
54    }
55}
56
57impl<X> StateHook<X> for ClearFsPropOnStartup
58where
59    X: XConn,
60{
61    fn call(&mut self, state: &mut State<X>, x: &mut X) -> Result<()> {
62        for id in x.existing_clients()? {
63            if !state.client_set.contains(&id) && x.client_should_be_managed(id) {
64                let net_wm_state = Atom::NetWmState.as_ref();
65                let full_screen = x.intern_atom(Atom::NetWmStateFullscreen.as_ref())?;
66                let mut wstate = match x.get_prop(id, net_wm_state) {
67                    Ok(Some(Prop::Cardinal(vals))) => vals,
68                    _ => vec![],
69                };
70                if wstate.contains(&full_screen) {
71                    wstate.retain(|&val| val != *full_screen);
72                    x.set_prop(id, net_wm_state, Prop::Cardinal(wstate))?;
73                }
74            }
75        }
76        Ok(())
77    }
78}
