16/// The possible valid actions to use when manipulating full screen state
17#[derive(Debug, Copy, Clone, PartialEq, Eq, PartialOrd, Ord)]
18pub enum FullScreenAction {
19    /// Force the window out of fullscreen state
20    Remove,
21    /// Force the window into fullscreen state
22    Add,
23    /// Toggle the fullscreen state of the window
24    Toggle,
25}
26
27/// Set the fullscreen state of a particular client
28pub fn set_fullscreen_state<X: XConn>(
29    id: Xid,
30    action: FullScreenAction,
31    state: &mut State<X>,
32    x: &X,
33) -> Result<()> {
34    use FullScreenAction::*;
35
36    let net_wm_state = Atom::NetWmState.as_ref();
37    let full_screen = x.intern_atom(Atom::NetWmStateFullscreen.as_ref())?;
38
39    let mut wstate = match x.get_prop(id, net_wm_state) {
40        Ok(Some(Prop::Cardinal(vals))) => vals,
41        _ => vec![],
42    };
43
44    let currently_fullscreen = wstate.contains(&full_screen);
45    debug!(%currently_fullscreen, ?action, %id, "setting fullscreen state");
46
47    if action == Add || (action == Toggle && !currently_fullscreen) {
48        let r = state
49            .client_set
50            .screen_for_client(&id)
51            .ok_or(Error::UnknownClient(id))?
52            .r;
53        state.client_set.float(id, r)?;
54        wstate.push(*full_screen);
55        x.set_client_config(id, &[ClientConfig::BorderPx(0)])?; // remove borders
56    } else if currently_fullscreen && (action == Remove || action == Toggle) {
57        state.client_set.sink(&id);
58        wstate.retain(|&val| val != *full_screen);
59        // replace borders
60        x.set_client_config(id, &[ClientConfig::BorderPx(state.config.border_width)])?;
61    }
62
63    x.set_prop(id, net_wm_state, Prop::Cardinal(wstate))?;
64    x.refresh(state)
65}

67/// Toggle the fullscreen state of the currently focused window.
68///
69/// **NOTE**: You will need to make use of [add_ewmh_hooks][0] for this action to
70///           work correctly.
71///
72///   [0]: crate::extensions::hooks::add_ewmh_hooks
73pub fn toggle_fullscreen<X: XConn>() -> Box<dyn KeyEventHandler<X>> {
74    key_handler(|state, x: &X| {
75        let id = match state.client_set.current_client() {
76            Some(&id) => id,
77            None => return Ok(()),
78        };
79
80        set_fullscreen_state(id, FullScreenAction::Toggle, state, x)
81    })
82}

84/// Jump to, or create a [Workspace][0].
85///
86/// Call 'get_name' to obtain a Workspace name and check to see if there is currently a Workspace
87/// with that name being managed by the WindowManager. If there is no existing workspace with the
88/// given name, create it with the supplied available layouts. If a matching Workspace _does_
89/// already exist then simply switch focus to it. This action is most useful when combined with the
90/// DefaultWorkspace hook that allows for auto populating named Workspaces when first focusing them.
91///
92/// > If you just want to dynamically select an existing workspace then you can use
93/// > [switch_to_workspace] to select from known workspace names.
94///
95///   [0]: crate::pure::Workspace
96pub fn create_or_switch_to_workspace<X>(
97    get_name: fn() -> Option<String>,
98    layouts: LayoutStack,
99) -> Box<dyn KeyEventHandler<X>>
100where
101    X: XConn,
102{
103    modify_with(move |cs| {
104        if let Some(name) = get_name() {
105            // if this errors it's because the tag is already present in the stackset
106            // so we can just focus it.
107            _ = cs.add_workspace(&name, layouts.clone());
108
109            cs.focus_tag(&name);
110        }
111    })
112}

114/// Jump to a [Workspace][0] by name.
115///
116/// Call 'select_name' to select a Workspace name and switch focus to it if it exists.
117///
118///   [0]: crate::pure::Workspace
119pub fn switch_to_workspace<X>(
120    select_name: fn(&[String]) -> Option<String>,
121) -> Box<dyn KeyEventHandler<X>>
122where
123    X: XConn,
124{
125    modify_with(move |cs| {
126        let tags = cs.ordered_tags();
127        if let Some(name) = select_name(&tags) {
128            cs.focus_tag(&name);
129        }
130    })
131}

133/// Focus a client with the given class as `WM_CLASS` or spawn the program with the given command
134/// if no such client exists.
135///
136/// This is useful for key bindings that are based on the program you want to work with rather than
137/// having to remember where things are running.
138pub fn focus_or_spawn<X>(class: &'static str, command: &'static str) -> Box<dyn KeyEventHandler<X>>
139where
140    X: XConn,
141{
142    key_handler(move |s: &mut State<X>, x: &X| {
143        let mut client = None;
144
145        for &id in s.client_set.clients() {
146            if let Some(Prop::UTF8String(classes)) = x.get_prop(id, Atom::WmClass.as_ref())? {
147                if classes.iter().any(|s| s == class) {
148                    client = Some(id);
149                    break;
150                }
151            }
152        }
153
154        x.modify_and_refresh(s, |cs| {
155            if let Some(id) = client {
156                cs.focus_client(&id)
157            } else if let Err(e) = spawn(command) {
158                error!(%e, %command, "unable to spawn program")
159            }
160        })
161    })
