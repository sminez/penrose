11/// Detect the current monitor set up and arrange the monitors if needed using [xrandr][1].
12///
13/// NOTE
14/// - Primary monitor will be set to `primary`
15/// - Monitor resolution is set using the --auto flag in xrandr
16/// - Only supports one and two monitor setups.
17///
18/// [1]: https://wiki.archlinux.org/index.php/Xrandr
19pub fn update_monitors_via_xrandr(
20    primary: &str,
21    secondary: &str,
22    position: RelativePosition,
23) -> Result<()> {
24    let raw = spawn_for_output("xrandr")?;
25    let status = raw
26        .lines()
27        .find(|line| line.starts_with(secondary))
28        .ok_or_else(|| {
29            Error::Custom("unable to find secondary monitor in xrandr output".to_owned())
30        })?
31        .split(' ')
32        .nth(1)
33        .ok_or_else(|| Error::Custom("unexpected xrandr output".to_owned()))?;
34
35    let pos = match position {
36        RelativePosition::Left => "--left-of",
37        RelativePosition::Right => "--right-of",
38        RelativePosition::Above => "--above",
39        RelativePosition::Below => "--below",
40    };
41
42    // force the primary monitor
43    spawn(format!("xrandr --output {} --primary --auto", primary))?;
44
45    // Updated the secondary monitor
46    match status {
47        "disconnected" => spawn(format!("xrandr --output {secondary} --off")),
48
49        "connected" => spawn(format!(
50            "xrandr --output {secondary} --auto {pos} {primary}",
51        )),
52
53        _ => Ok(()),
54    }
55}

57/// A notification level when calling notify-send
58///
59/// The effect of this on the generated notification will depend on the notification
60/// daemon that you are using.
61#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
62pub enum NotifyLevel {
63    /// Low priority
64    Low,
65    /// Normal priority
66    Normal,
67    /// Critical priority
68    Critical,
69}
70
71/// Send a notification using the `notify-send` command line program
72pub fn notify_send(title: impl AsRef<str>, body: impl AsRef<str>) -> Result<()> {
73    notify_send_custom(title, body, NotifyLevel::Normal, 5000)
74}
75
76/// Send a notification using the `notify-send` command line program
77///
78/// Duration is in ms.
79pub fn notify_send_custom(
80    title: impl AsRef<str>,
81    body: impl AsRef<str>,
82    level: NotifyLevel,
83    duration: usize,
84) -> Result<()> {
85    let level = match level {
86        NotifyLevel::Low => "low",
87        NotifyLevel::Normal => "normal",
88        NotifyLevel::Critical => "critical",
89    };
90
91    spawn_with_args(
92        "notify-send",
93        &[
94            "-u",
95            level,
96            "-t",
97            &duration.to_string(),
98            title.as_ref(),
99            body.as_ref(),
100        ],
101    )
102}
