11/// The result of attempting to match against user input in a [DMenu]
12#[derive(Debug, Clone)]
13pub enum MenuMatch {
14    /// The selected line along its line number (0 indexed)
15    Line(usize, String),
16    /// Nothing matched and this was the user's input when they hit Return
17    UserInput(String),
18    /// The user exited out of matching or had nothing typed
19    NoMatch,
20}
21
22/// Two different derivatives of dmenu
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub enum DMenuKind {
25    /// Suckless's version of dmenu
26    ///
27    /// <https://tools.suckless.org/dmenu/>
28    Suckless,
29    /// Newer `dmenu-rs`
30    ///
31    /// <https://github.com/Shizcow/dmenu-rs>
32    Rust,
33}

35/// Custom configuration options for [`DMenu`].
36///
37/// # Example
38/// ```no_run
39/// # use penrose::extensions::util::dmenu::*;
40/// let dc = DMenuConfig {
41///     show_line_numbers: true,
42///     show_on_bottom: true,
43///     password_input: true,
44///     ignore_case: false,
45///     n_lines: 0,
46///     custom_font: Some("JetBrains Nerd Font Mono".to_owned()),
47///     kind: DMenuKind::Rust,
48///     custom_prompt: Some(" ".to_owned()),
49///     ..DMenuConfig::default()
50/// };
51/// ```
52#[allow(clippy::struct_excessive_bools)]
53#[derive(Debug, Clone)]
54pub struct DMenuConfig {
55    /// Should line numbers be displayed to the user?
56    ///
57    /// Default: false
58    pub show_line_numbers: bool,
59
60    /// Show dmenu at the bottom the the screen.
61    ///
62    /// Default: false
63    pub show_on_bottom: bool,
64
65    /// Should dmenu treat the input as a password and render characters as '*'?
66    ///
67    /// > This requires the [Password][1] patch in order to work.
68    /// > or in the case of dmenu-rs it requires the password plugin.
69    ///
70    /// Default: false
71    ///
72    /// [1]: https://tools.suckless.org/dmenu/patches/password/
73    /// [1]: https://github.com/Shizcow/dmenu-rs/tree/master/src/plugins
74    pub password_input: bool,
75
76    /// Should dmenu ignore case in the user input when matching?
77    ///
78    /// Default: false
79    pub ignore_case: bool,
80
81    /// Background color for the rendered window
82    ///
83    /// Default: #282828
84    pub bg_color: Color,
85
86    /// Foreground color for text
87    ///
88    /// Default: #ebdbb2
89    pub fg_color: Color,
90
91    /// Selected line background color
92    ///
93    /// Default: #458588
94    pub selected_color: Color,
95
96    /// Number of lines to display at a time.
97    ///
98    /// Setting n_lines=0 will result in the choices being displayed horizontally
99    /// instead of vertically.
100    ///
101    /// Default: 10
102    pub n_lines: u8,
103
104    /// Allow the user to load a custom font
105    ///
106    /// Default: None
107    pub custom_font: Option<String>,
108
109    /// Specify to kind of dmenu to use
110    ///
111    /// Default: Suckless
112    pub kind: DMenuKind,
113
114    /// Optional prompt customization.
115    ///
116    /// Default: None
117    pub custom_prompt: Option<String>,
118}

138impl DMenuConfig {
139    /// Construct a default config with a custom prompt
140    pub fn with_prompt(prompt: &str) -> Self {
141        Self {
142            custom_prompt: Some(prompt.to_string()),
143            ..Default::default()
144        }
145    }
146
147    /// Build the dmenu flags
148    fn flags(&self, screen_index: usize) -> Vec<String> {
149        let Self {
150            show_on_bottom,
151            password_input,
152            ignore_case,
153            bg_color,
154            fg_color,
155            selected_color,
156            n_lines,
157            custom_font,
158            kind,
159            custom_prompt,
160            ..
161        } = self;
162
163        // Only some command line options require the "--" for the rust version.
164        let prefix = match kind {
165            DMenuKind::Suckless => "-",
166            DMenuKind::Rust => "--",
167        };
168
169        let mut flags = vec!["-m".to_owned(), screen_index.to_string()];
170
171        flags.push(format!("{prefix}nb"));
172        flags.push(bg_color.as_rgb_hex_string());
173
174        flags.push(format!("{prefix}nf"));
175        flags.push(fg_color.as_rgb_hex_string());
176
177        flags.push(format!("{prefix}sb"));
178        flags.push(selected_color.as_rgb_hex_string());
179
180        if *n_lines > 0 {
181            flags.push("-l".to_owned());
182            flags.push(n_lines.to_string());
183        }
184
185        if *show_on_bottom {
186            flags.push("-b".to_owned());
187        }
188
189        if *password_input {
190            flags.push("-P".to_owned());
191        }
192
193        if *ignore_case {
194            flags.push("-i".to_owned());
195        }
196
197        if let Some(font) = custom_font {
198            flags.push(format!("{prefix}fn"));
199            flags.push(font.to_owned());
200        }
201
202        if let Some(prompt) = custom_prompt {
203            flags.push("-p".to_owned());
204            flags.push(prompt.to_owned());
205        }
206
207        flags
208    }
209}

211/// A wrapper around the suckless [dmenu](http://tools.suckless.org/dmenu/) program
212/// for creating dynamic menus in penrose.
213#[derive(Debug, Clone)]
214pub struct DMenu {
215    /// Holds the custom dmenu configuration for this instance.
216    config: DMenuConfig,
217    /// The screen index this instance of dmenu will show up on
218    screen_index: usize,
219}
220
221impl DMenu {
222    /// Create a new [`DMenu`] command which can be triggered and reused by calling
223    /// the `run` method for a basic dmenu prompt, or the `build_menu`
224    /// for more advanced selection menus.
225    pub fn new(config: &DMenuConfig, screen_index: usize) -> Self {
226        Self {
227            config: config.to_owned(),
228            screen_index,
229        }
230    }
231
232    /// Used for launching regular old [`DMenu`] with no menu matching
233    /// via the `dmenu_run` wrapper script.
234    pub fn run(&self) -> Result<()> {
235        let args = self.config.flags(self.screen_index);
236        let spawned_process = Command::new("dmenu_run").args(args).spawn();
237
238        match spawned_process {
239            Ok(mut process) => match process.wait() {
240                Ok(_) => Ok(()),
241                Err(e) => Err(e.into()),
242            },
243            Err(e) => Err(e.into()),
244        }
245    }
246
247    /// Run this [`DMenu`] command and return the selected choice.
248    ///
249    /// # Example
250    /// ```no_run
251    /// # use penrose::extensions::util::dmenu::*;
252    /// let screen_index = 0;
253    /// let dmenu = DMenu::new(&DMenuConfig::default(), screen_index);
254    ///
255    /// let choices = vec!["some", "choices", "to", "pick", "from"];
256    ///
257    /// match dmenu.build_menu(choices).unwrap() {
258    ///     MenuMatch::Line(i, s) => println!("matched '{}' on line '{}'", s, i),
259    ///     MenuMatch::UserInput(s) => println!("user input: '{}'", s),
260    ///     MenuMatch::NoMatch => println!("no match"),
261    /// }
262    /// ```
263    // #[allow(clippy::pattern_type_mismatch)]
264    pub fn build_menu(&self, param_choices: Vec<impl Into<String>>) -> Result<MenuMatch> {
265        let choices: Vec<String> = param_choices
266            .into_iter()
267            .map(std::convert::Into::into)
268            .collect();
269        let raw = self.raw_user_choice_from_dmenu(&choices)?;
270        let choice = raw.trim();
271
272        if choice.is_empty() {
273            return Ok(MenuMatch::NoMatch);
274        }
275
276        let res = choices
277            .iter()
278            .enumerate()
279            .find(|(i, s)| {
280                if self.config.show_line_numbers {
281                    format!("{i:<3} {s}") == choice
282                } else {
283                    *s == choice
284                }
285            })
286            .map_or_else(
287                || MenuMatch::UserInput(choice.to_owned()),
288                |(i, _)| {
289                    MenuMatch::Line(
290                        i,
291                        choices.get(i).expect("Indexing choices panicked").clone(),
292                    )
293                },
294            );
295
296        Ok(res)
297    }
298
299    /// Get a vector of choices as bytes
300    fn choices_as_input_bytes(&self, choices: &[String]) -> Vec<u8> {
301        if self.config.show_line_numbers {
302            choices
303                .iter()
304                .enumerate()
305                .map(|(i, s)| format!("{i:<3} {s}"))
306                .collect::<Vec<String>>()
307                .join("\n")
308                .as_bytes()
309                .to_vec()
310        } else {
311            choices.join("\n").as_bytes().to_vec()
312        }
313    }
314
315    /// Launch a shell process with all arguments to dmenu
316    fn raw_user_choice_from_dmenu(&self, choices: &[String]) -> Result<String> {
317        let args = self.config.flags(self.screen_index);
318        let mut proc = Command::new("dmenu")
319            .stdin(Stdio::piped())
320            .stdout(Stdio::piped())
321            .args(args)
322            .spawn()?;
323
324        {
325            // Taking stdin here and dropping it when the block scope ends to close it and
326            // let dmenu determine the end of input
327            let mut stdin = proc
328                .stdin
329                .take()
330                .ok_or_else(|| Error::Custom("unable to open stdin".to_owned()))?;
331
332            stdin.write_all(&self.choices_as_input_bytes(choices))?;
333        }
334
335        let mut raw = String::new();
336        proc.stdout
337            .ok_or_else(|| Error::Custom("failed to spawn dmenu".to_owned()))?
338            .read_to_string(&mut raw)?;
339
340        Ok(raw)
341    }
342}

344#[cfg(test)]
345mod tests {
346    use super::{DMenuConfig, DMenuKind};

348    /// Flags [ nb, nf, sb, and nf] need to be modified for the different
349    /// versions of dmenu. Classic Suckless dmenu uses a single dash "-", dmenu-rs
350    /// uses the more modern cli style of double dashes "--".
351    /// This test depends on the order the flags are loaded into the array, so if the order
352    /// is changed in the flags function, these tests will fail. There is a better way, but
353    /// this works for now.
354    #[test]
355    fn dmenu_suckless_config_test() {
356        let dc = DMenuConfig {
357            custom_font: Some("mono".to_owned()),
358            ..DMenuConfig::default()
359        };
360
361        // Should default to suckless c-style dmenu
362        assert_eq!(dc.kind, DMenuKind::Suckless);
363        let flags = dc.flags(0);
364
365        for (i, flag) in flags.into_iter().enumerate() {
366            if i == 2 {
367                assert_eq!(flag, "-nb".to_owned());
368            }
369            if i == 4 {
370                assert_eq!(flag, "-nf".to_owned());
371            }
372            if i == 6 {
373                assert_eq!(flag, "-sb".to_owned());
374            }
375            if i == 10 {
376                assert_eq!(flag, "-fn".to_owned());
377            }
378        }
379    }
