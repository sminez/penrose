1//! Built-in hooks
2use crate::{
3    core::{State, WinId, conn::Conn, hooks::LayoutHook},
4    pure::geometry::Rect,
5};
6
7/// Simple gaps around the window placement of the enclosed [Layout][crate::core::layout::Layout].
8///
9/// `outer_px` controls the width of the gap around the edge of the screen and `inner_px`
10/// controls the gap around each individual window. Set both equal to one another to have
11/// a consistant gap size in all places.
12#[derive(Debug, Clone, Default)]
13pub struct SpacingHook {
14    /// The desired outer gap size in pixels
15    pub outer_px: u32,
16    /// The desired inner gap size in pixels
17    pub inner_px: u32,
18    /// The number of pixels to reserve at the top of the screen
19    pub top_px: u32,
20    /// The number of pixels to reserve at the bottom of the screen
21    pub bottom_px: u32,
22}
23
24impl<C: Conn> LayoutHook<C> for SpacingHook {
25    fn transform_initial(&mut self, mut r: Rect, _: &State<C>, _: &mut C) -> Rect {
26        if r.w == 0 || r.h == 0 {
27            return r;
28        }
29
30        r.y += self.top_px as i32;
31        r.h = r.h - self.top_px - self.bottom_px;
32
33        shrink(r, self.outer_px)
34    }
35
36    fn transform_positions(
37        &mut self,
38        _: Rect,
39        positions: Vec<(WinId, Rect)>,
40        _: &State<C>,
41        _: &mut C,
42    ) -> Vec<(WinId, Rect)> {
43        positions
44            .into_iter()
45            .map(|(id, r)| (id, shrink(r, self.inner_px)))
46            .collect()
47    }
48}
49
50fn shrink(r: Rect, px: u32) -> Rect {
51    if r.w == 0 || r.h == 0 {
52        return r;
53    }
54
55    Rect {
56        x: r.x + px as i32,
57        y: r.y + px as i32,
58        w: r.w - 2 * px,
59        h: r.h - 2 * px,
60    }
61}
