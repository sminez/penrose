1//! Built-in layout transformers.

8simple_transformer!(
9    /// Wrap an existing layout and reflect its window positions horizontally.
10    ReflectHorizontal,
11    reflect_horizontal,
12    "Reflected"
13);
14
15fn reflect_horizontal(r_s: Rect, positions: Vec<(Xid, Rect)>) -> Vec<(Xid, Rect)> {
16    let offset = (r_s.x + r_s.w as i32 / 2) * 2;
17
18    positions
19        .into_iter()
20        .map(|(id, mut r)| {
21            r.x = offset.saturating_sub(r.x).saturating_sub(r.w as i32);
22            (id, r)
23        })
24        .collect()
25}

27simple_transformer!(
28    /// Wrap an existing layout and reflect its window positions vertically.
29    ReflectVertical,
30    reflect_vertical,
31    "Flipped"
32);
33
34fn reflect_vertical(r: Rect, positions: Vec<(Xid, Rect)>) -> Vec<(Xid, Rect)> {
35    let offset = (r.y + r.h as i32 / 2) * 2;
36
37    positions
38        .into_iter()
39        .map(|(id, mut r)| {
40            r.y = offset.saturating_sub(r.y).saturating_sub(r.h as i32);
41            (id, r)
42        })
43        .collect()
44}

46/// Simple gaps around the window placement of the enclosed [Layout].
47///
48/// `outer_px` controls the width of the gap around the edge of the screen and `inner_px`
49/// controls the gap around each individual window. Set both equal to one another to have
50/// a consistant gap size in all places.
51#[derive(Debug, Clone)]
52pub struct Gaps {
53    /// The inner [Layout] having gaps applied to it.
54    pub layout: Box<dyn Layout>,
55    /// The desired outer gap size in pixels
56    pub outer_px: u32,
57    /// The desired inner gap size in pixels
58    pub inner_px: u32,
59}
60
61impl Gaps {
62    /// Wrap an existing [Layout] with the given gap sizes.
63    pub fn wrap(layout: Box<dyn Layout>, outer_px: u32, inner_px: u32) -> Box<dyn Layout> {
64        Box::new(Self {
65            layout,
66            outer_px,
67            inner_px,
68        })
69    }
70}
71
72fn shrink(r: Rect, px: u32) -> Rect {
73    if r.w == 0 || r.h == 0 {
74        return r;
75    }
76
77    Rect {
78        x: r.x + px as i32,
79        y: r.y + px as i32,
80        w: r.w - 2 * px,
81        h: r.h - 2 * px,
82    }
83}
84
85impl LayoutTransformer for Gaps {
86    fn transformed_name(&self) -> String {
87        self.layout.name()
88    }
89
90    fn inner_mut(&mut self) -> &mut Box<dyn Layout> {
91        &mut self.layout
92    }
93
94    fn transform_initial(&self, r: Rect) -> Rect {
95        shrink(r, self.outer_px)
96    }
97
98    fn transform_positions(&mut self, _: Rect, positions: Vec<(Xid, Rect)>) -> Vec<(Xid, Rect)> {
99        positions
100            .into_iter()
101            .map(|(id, r)| (id, shrink(r, self.inner_px)))
102            .collect()
103    }
104}

106/// Reserve `px` pixels at the top of the screen.
107///
108/// Typically used for providing space for a status bar.
109#[derive(Debug, Clone)]
110pub struct ReserveTop {
111    /// The wrapped inner layout
112    pub layout: Box<dyn Layout>,
113    /// The number of pixels to reserve at the top of the screen
114    pub px: u32,
115}
116
117impl ReserveTop {
118    /// Wrap an existing [Layout] with the given reserved area.
119    pub fn wrap(layout: Box<dyn Layout>, px: u32) -> Box<dyn Layout> {
120        Box::new(Self { layout, px })
121    }
122}
123
124impl LayoutTransformer for ReserveTop {
125    fn transformed_name(&self) -> String {
126        self.layout.name()
127    }
128
129    fn inner_mut(&mut self) -> &mut Box<dyn Layout> {
130        &mut self.layout
131    }
132
133    fn transform_initial(&self, mut r: Rect) -> Rect {
134        if r.w == 0 || r.h == 0 {
135            return r;
136        }
137
138        r.y += self.px as i32;
139        r.h -= self.px;
140
141        r
142    }
143}
