1//! Built-in layout messages.
2//!
3//! It is not a hard requirement for [Layout][0] implementations to handle each
4//! of the messages provided by this module but wherever possible you should
5//! attempt to do so if the semantics of the message make sense for the
6//! layout you are writing.
7//!
8//!   [0]: crate::core::layout::Layout
9
10/// Mark a type as being usable as a [Message][0] for sending to a [Layout][1]
11/// ```
12/// use penrose::impl_message;
13///
14/// struct MyMessageType {
15///     important_data: u8
16/// }
17///
18/// impl_message!(MyMessageType);
19/// ```
20///
21///   [0]: crate::core::layout::Message
22///   [1]: crate::core::layout::Layout
23#[macro_export]
24macro_rules! impl_message {
25    ($m:ident) => {
26        impl $crate::core::layout::IntoMessage for $m {}
27    };
28}
