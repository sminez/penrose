10/// Know property types that should be returnable by XConn impls when they check
11/// window properties.
12#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
13#[derive(Debug, PartialEq, Eq, Hash, Clone)]
14pub enum Prop {
15    /// One or more X Atoms
16    Atom(Vec<String>),
17    /// Raw bytes for when the prop type is non-standard
18    Bytes(Vec<u32>),
19    /// A cardinal number
20    Cardinal(Vec<u32>),
21    /// UTF-8 encoded string data
22    UTF8String(Vec<String>),
23    /// An X window IDs
24    Window(Vec<Xid>),
25    /// The WmHints properties for this window
26    WmHints(WmHints),
27    /// The WmNormalHints properties for this window
28    WmNormalHints(WmNormalHints),
29}

46        /// Icon mask hint is set
47        const ICON_MASK_HINT     = 0b0000100000;
48        /// Window group hint is set
49        const WINDOW_GROUP_HINT  = 0b0001000000;

51        /// Urgency hint is set

160            icon_win,
161            icon_position,
162            icon_mask,
163            window_group,

167    /// Try to construct a [WmHints] instance from raw bytes.
168    ///
169    /// This method expects a slice of 9 u32s corresponding to the C struct layout shown below.
170    ///
171    /// ```C
172    /// typedef struct {
173    ///     long flags;          /* marks which fields in this structure are defined */
174    ///     Bool input;          /* does this application rely on the window manager to
175    ///                             get keyboard input? */
176    ///     int initial_state;   /* see below */
177    ///     Pixmap icon_pixmap;  /* pixmap to be used as icon */
178    ///     Window icon_window;  /* window to be used as icon */
179    ///     int icon_x, icon_y;  /* initial position of icon */
180    ///     Pixmap icon_mask;    /* pixmap to be used as mask for icon_pixmap */
181    ///     XID window_group;    /* id of related window group */
182    ///     /* this structure may be extended in the future */
183    /// } XWMHints;
184    /// ```
185    pub fn try_from_bytes(raw: &[u32]) -> Result<Self> {
186        if raw.len() != 9 {
187            return Err(Error::InvalidHints {
188                reason: format!(
189                    "raw bytes should be [u32; 9] for WmHints, got [u32; {}]",
190                    raw.len()
191                ),
192            });
193        }
194
195        let flags = WmHintsFlags::from_bits(raw[0]).unwrap();
196        let accepts_input = !flags.contains(WmHintsFlags::INPUT_HINT) || raw[1] > 0;
197        let initial_state = match (flags.contains(WmHintsFlags::STATE_HINT), raw[2]) {
198            (true, 0) => WmState::Withdrawn,
199            (true, 1) | (false, _) => WmState::Normal,
200            (true, 2) => WmState::Iconic,
201            _ => {
202                return Err(Error::InvalidHints {
203                    reason: format!("initial state flag should be 0, 1, 2: got {}", raw[2]),
204                })
205            }
206        };
207
208        Ok(Self {
209            flags,
210            accepts_input,
211            initial_state,
212            icon_pixmap: raw[3],
213            icon_win: Xid(raw[4]),
214            icon_position: Point::new(raw[5] as i32, raw[6] as i32),
215            icon_mask: raw[7],
216            window_group: raw[8],
217        })
218    }
219}

221/// Client requested hints about window geometry.
222///
223/// See the ICCCM [spec][1] for further details or the [Xlib manual][2] for more details of the
224/// data fromat but note that Penrose does not honour the following hints:
225///   - gravity
226///   - increment
227///   - aspect ratio
228///
229/// [1]: https://www.x.org/releases/X11R7.6/doc/xorg-docs/specs/ICCCM/icccm.html#wm_normal_hints_property
230/// [2]: https://tronche.com/gui/x/xlib/ICC/client-to-window-manager/wm-normal-hints.html
231#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
232#[derive(Debug, PartialEq, Eq, Hash, Clone)]
233pub struct WmNormalHints {
234    pub(crate) flags: WmNormalHintsFlags,
235    pub(crate) base: Option<Rect>,
236    pub(crate) min: Option<Rect>,
237    pub(crate) max: Option<Rect>,
238    pub(crate) user_specified: Option<Rect>,
239}

241impl WmNormalHints {

244        flags: WmNormalHintsFlags,
245        base: Option<Rect>,
246        min: Option<Rect>,
247        max: Option<Rect>,
248        user_specified: Option<Rect>,
249    ) -> Self {
250        Self {
251            flags,
252            base,
253            min,
254            max,
255            user_specified,
256        }
257    }
258
259    /// Apply these size hints to a given [Rect].
260    ///
261    /// > Currently only the max size is respected
262    pub fn apply_to(&self, mut r: Rect) -> Rect {
263        if let Some(max) = self.max {
264            if r.is_larger_than(&max) {
265                r.w = max.w;
266                r.h = max.h
267            }
268        }
269
270        if let Some(min) = self.min {
271            if min.is_larger_than(&r) {
272                r.w = min.w;
273                r.h = min.h;
274            }
275        }
276
277        r
278    }

280    /// Try to construct a [WmNormalHints] instance from raw bytes.
281    ///
282    /// This method expects a slice of 18 u32s corresponding to the C struct layout shown below.
283    ///
284    /// ```C
285    /// typedef struct {
286    ///     long flags;                /* marks which fields in this structure are defined */
287    ///     int x, y;                  /* Obsolete */
288    ///     int width, height;         /* Obsolete */
289    ///     int min_width, min_height;
290    ///     int max_width, max_height;
291    ///     int width_inc, height_inc;
292    ///     struct {
293    ///            int x;              /* numerator */
294    ///            int y;              /* denominator */
295    ///     } min_aspect, max_aspect;
296    ///     int base_width, base_height;
297    ///     int win_gravity;
298    ///     /* this structure may be extended in the future */
299    /// } XSizeHints;
300    /// ```
301    pub fn try_from_bytes(raw: &[u32]) -> Result<Self> {
302        if raw.len() != 18 {
303            return Err(Error::InvalidHints {
304                reason: format!(
305                    "raw bytes should be [u32; 18] for WmNormalHints, got [u32; {}]",
306                    raw.len()
307                ),
308            });
309        }
310
311        let flags = WmNormalHintsFlags::from_bits(raw[0]).unwrap();
312
313        // These properties are marked as obsolete but some clients still set them
314        // so it they are useful as fallbacks
315        let (x, y) = (raw[1] as i32, raw[2] as i32);
316        let (user_w, user_h) = (raw[3], raw[4]);
317
318        let (min_w, min_h) = (raw[5], raw[6]);
319        let (max_w, max_h) = (raw[7], raw[8]);
320        let (base_w, base_h) = (raw[15], raw[16]);
321
322        // ignoring increment, aspect ratio, gravity as they are not used in
323        // the main WindowManager logic
324
325        let if_set = |x, y, w, h| {
326            if w > 0 && h > 0 {
327                Some(Rect::new(x, y, w, h))
328            } else {
329                None
330            }
331        };
332
333        Ok(Self {
334            flags,
335            base: if_set(x, y, base_w, base_h),
336            min: if_set(x, y, min_w, min_h),
337            max: if_set(x, y, max_w, max_h),
338            user_specified: if_set(x, y, user_w, user_h),
339        })
340    }
341}

343/// Window Attributes honoured by penose.
344///
345/// Only a small subset of window attributes are checked and honoured by penrose. This list may be
346/// extended in future.
347///
348/// ```C
349/// typedef struct xcb_get_window_attributes_reply_t {
350///     uint8_t        response_type;
351///     uint8_t        backing_store;
352///     uint16_t       sequence;
353///     uint32_t       length;
354///     xcb_visualid_t visual;
355///     uint16_t       _class;
356///     uint8_t        bit_gravity;
357///     uint8_t        win_gravity;
358///     uint32_t       backing_planes;
359///     uint32_t       backing_pixel;
360///     uint8_t        save_under;
361///     uint8_t        map_is_installed;
362///     uint8_t        map_state;
363///     uint8_t        override_redirect;
364///     xcb_colormap_t colormap;
365///     uint32_t       all_event_masks;
366///     uint32_t       your_event_mask;
367///     uint16_t       do_not_propagate_mask;
368///     uint8_t        pad0[2];
369/// } xcb_get_window_attributes_reply_t;
370/// ```
371#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
372#[derive(Debug, PartialEq, Eq, Hash, Clone)]
373pub struct WindowAttributes {
374    pub(crate) override_redirect: bool,
375    pub(crate) map_state: MapState,
376    pub(crate) window_class: WindowClass,
377}
