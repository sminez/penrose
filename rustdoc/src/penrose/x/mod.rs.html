83/// A handle on a running X11 connection that we can use for issuing X requests.
84///
85/// XConn is intended as an abstraction layer to allow for communication with the underlying
86/// display system (assumed to be X) using whatever mechanism the implementer wishes. In theory, it
87/// should be possible to write an implementation that allows penrose to run on systems not using X
88/// as the windowing system but X idioms and high level event types / client interations are
89/// assumed.
90pub trait XConn: Send {
91    /// The ID of the window manager root window.
92    fn root(&mut self) -> WinId;
93    /// Ask the X server for the dimensions of each currently available screen.
94    fn screen_details(&mut self) -> Result<Vec<Rect>>;
95    /// Ask the X server for the current (x, y) coordinate of the mouse cursor.
96    fn cursor_position(&mut self) -> Result<Point>;
97
98    /// Grab the specified key and mouse states, intercepting them for processing within
99    /// the window manager itself.
100    fn grab(&mut self, key_codes: &[KeyCode], mouse_states: &[MouseState]) -> Result<()>;
101    /// Block and wait for the next event from the X server so it can be processed.
102    fn next_event(&mut self) -> Result<XEvent>;
103    /// Flush any pending events to the X server.
104    fn flush(&mut self);
105
106    /// Look up the [WinId] of a given [Atom] name. If it is not currently interned, intern it.
107    fn intern_atom(&mut self, atom: &str) -> Result<WinId>;
108    /// Look up the string name of a given [Atom] by its [WinId].
109    fn atom_name(&mut self, xid: WinId) -> Result<String>;
110
111    /// Look up the current dimensions and position of a given client window.
112    fn client_geometry(&mut self, client: WinId) -> Result<Rect>;
113    /// Ask the X server for the IDs of all currently known client windows
114    fn existing_clients(&mut self) -> Result<Vec<WinId>>;
115
116    /// Map the given client window to the screen with its current geometry, making it visible.
117    fn map(&mut self, client: WinId) -> Result<()>;
118    /// Unmap the given client window from the screen, hiding it.
119    fn unmap(&mut self, client: WinId) -> Result<()>;
120    /// Kill the given client window, closing it.
121    fn kill(&mut self, client: WinId) -> Result<()>;
122    /// Set X input focus to be held by the given client window.
123    fn focus(&mut self, client: WinId) -> Result<()>;
124
125    /// Look up a specific property on a given client window.
126    fn get_prop(&mut self, client: WinId, prop_name: &str) -> Result<Option<Prop>>;
127    /// List the known property names set for a given client.
128    fn list_props(&mut self, client: WinId) -> Result<Vec<String>>;
129    /// Get the current [WmState] for a given client window.
130    fn get_wm_state(&mut self, client: WinId) -> Result<Option<WmState>>;
131    /// Request the [WindowAttributes] for a given client window from the X server.
132    fn get_window_attributes(&mut self, client: WinId) -> Result<WindowAttributes>;
133
134    /// Set the current [WmState] for a given client window.
135    fn set_wm_state(&mut self, client: WinId, wm_state: WmState) -> Result<()>;
136    /// Set a specific property on a given client window.
137    fn set_prop(&mut self, client: WinId, name: &str, val: Prop) -> Result<()>;
138    /// Delete a property for a given client window.
139    fn delete_prop(&mut self, client: WinId, prop_name: &str) -> Result<()>;
140    /// Set one or more [ClientAttr] for a given client window.
141    fn set_client_attributes(&mut self, client: WinId, attrs: &[ClientAttr]) -> Result<()>;
142    /// Set the [ClientConfig] for a given client window.
143    fn set_client_config(&mut self, client: WinId, data: &[ClientConfig]) -> Result<()>;
144    /// Send a [ClientMessage] to a given client.
145    fn send_client_message(&mut self, msg: ClientMessage) -> Result<()>;
146
147    /// Reposition the mouse cursor to the given (x, y) coordinates within the specified window.
148    /// This method should not be called directly: use `warp_pointer_to_window` or `warp_pointer_to_screen`
149    /// instead.
150    fn warp_pointer(&mut self, id: WinId, x: i16, y: i16) -> Result<()>;
151}

159/// The check for whether or not we manage an _existing_ client is a little different from
160/// whether or not we manage a _new_ client.
161fn existing_client_should_be_managed<X: XConn>(x: &mut X, id: WinId) -> bool {
162    let attrs = match x.get_window_attributes(id) {
163        Ok(attrs) => attrs,
164        _ => {
165            warn!(%id, "unable to pull window attributes for client: skipping.");
166            return false;
167        }
168    };
169
170    let wm_state = match x.get_wm_state(id) {
171        Ok(state) => state,
172        _ => {
173            warn!(%id, "unable to pull wm state for client: skipping.");
174            return false;
175        }
176    };
177
178    info!(%id, ?attrs, ?wm_state, "processing client");
179
180    let WindowAttributes {
181        override_redirect,
182        map_state,
183        ..
184    } = attrs;
185
186    let viewable = map_state == MapState::Viewable;
187    let iconic = wm_state == Some(WmState::Iconic);
188
189    // This condition for determining what windows we should manage is
190    // taken from the `scan` function found in both dwm and XMonad.
191    !override_redirect && (viewable || iconic)
192}

224    fn handle_event(
225        &mut self,
226        evt: Self::Event,
227        key_bindings: &mut KeyBindings<Self>,
228        mouse_bindings: &mut MouseBindings<Self>,
229        state: &mut State<Self>,
230    ) -> Result<()> {
231        use XEvent::*;
232
233        match evt {
234            ClientMessage(m) => handle::client_message(m.clone(), state, self)?,
235            ConfigureNotify(e) if e.is_root => handle::detect_screens(state, self)?,
236            ConfigureNotify(_) => (), // Not currently handled
237            ConfigureRequest(e) => handle::configure_request(&e, state, self)?,
238            Enter(p) => handle::enter(p, state, self)?,
239            Expose(_) => (), // Not currently handled
240            FocusIn(id) => handle::focus_in(id, state, self)?,
241            Destroy(id) => handle::destroy(id, state, self)?,
242            KeyPress(code) => handle::keypress(code, key_bindings, state, self)?,
243            Leave(p) => handle::leave(p, state, self)?,
244            MappingNotify => handle::mapping_notify(key_bindings, mouse_bindings, self)?,
245            MapRequest(xid) => handle::map_request(xid, state, self)?,
246            MouseEvent(e) => handle::mouse_event(e.clone(), mouse_bindings, state, self)?,
247            MotionNotify(e) => handle::motion_event(e.clone(), mouse_bindings, state, self)?,
248            PropertyNotify(_) => (), // Not currently handled
249            RandrNotify => handle::detect_screens(state, self)?,
250            ScreenChange => handle::screen_change(state, self)?,
251            UnmapNotify(id) => handle::unmap_notify(id, state, self)?,
252
253            _ => (), // XEvent is non-exhaustive
254        }
255
256        Ok(())
257    }

289    fn position_client(&mut self, id: WinId, mut r: Rect) -> Result<()> {
290        let p = Atom::WmNormalHints.as_ref();
291        if let Ok(Some(Prop::WmNormalHints(hints))) = self.get_prop(id, p) {
292            trace!(%id, ?hints, "client has WmNormalHints: applying size hints");
293            r = hints.apply_to(r);
294        }
295
296        trace!(%id, ?r, "positioning client");
297        self.set_client_config(id, &[ClientConfig::Position(r)])
298    }

300    fn show_client(&mut self, id: WinId, state: &mut State<Self>) -> Result<()> {
301        self.set_wm_state(id, WmState::Normal)?;
302        self.map(id)?;
303
304        if state.client_set.contains(&id) {
305            state.conn_state.mapped.insert(id);
306        }
307
308        Ok(())
309    }

311    fn hide_client(&mut self, id: WinId, state: &mut State<Self>) -> Result<()> {
312        if !state.conn_state.mapped.contains(&id) {
313            return Ok(());
314        }
315
316        self.set_client_attributes(id, &[ClientAttr::ClientUnmapMask])?;
317        self.unmap(id)?;
318        self.set_client_attributes(id, &[ClientAttr::ClientEventMask])?;
319        self.set_wm_state(id, WmState::Iconic)?;
320
321        state.conn_state.mapped.remove(&id);
322        state
323            .conn_state
324            .pending_unmap
325            .entry(id)
326            .and_modify(|count| *count += 1)
327            .or_insert(1);
328
329        Ok(())
330    }

370    fn client_should_float(&mut self, id: WinId, floating_classes: &[String]) -> bool {
371        trace!(%id, "fetching WmClass prop");
372        if let Ok(Some(Prop::UTF8String(strs))) = self.get_prop(id, Atom::WmClass.as_ref())
373            && strs.iter().any(|c| floating_classes.contains(c))
374        {
375            debug!(%id, ?floating_classes, "window has a floating class: setting to floating state");
376            return true;
377        }
378
379        trace!(%id, "fetching NetWmWindowType prop");
380        let window_types = match self.get_prop(id, Atom::NetWmWindowType.as_ref()) {
381            Ok(tys) => tys,
382            _ => return false,
383        };
384        debug!(?window_types, "id window types");
385
386        let float_types: Vec<&str> = AUTO_FLOAT_WINDOW_TYPES.iter().map(|a| a.as_ref()).collect();
387
388        match window_types {
389            Some(Prop::Atom(atoms)) => atoms.iter().any(|a| float_types.contains(&a.as_ref())),
390            _ => false,
391        }
392    }

405        let net_wm_state = Atom::NetWmState.as_ref();

407            Ok(atom) => atom,
408            _ => return false,

411        let wstate = match self.get_prop(id, net_wm_state) {
412            Ok(Some(Prop::Cardinal(vals))) => vals,
413            _ => vec![],

416        wstate.contains(&full_screen)

449    fn restack<'a, I>(&mut self, mut ids: I) -> Result<()>
450    where
451        WinId: 'a,
452        I: Iterator<Item = &'a WinId>,
453    {
454        let mut previous = match ids.next() {
455            Some(id) => *id,
456            None => return Ok(()), // nothing to stack
457        };
458
459        for &id in ids {
460            self.set_client_config(id, &[ClientConfig::StackAbove(previous)])?;
461            previous = id;
462        }
463
464        Ok(())
465    }

467    // A "best effort" attempt to manage existing clients on the workspaces they were present
468    // on previously. This is not guaranteed to preserve the stack order or correctly handle
469    // any clients that were on invisible workspaces / workspaces that no longer exist.
470    //
471    // NOTE: the check for if each client is already in state is in case a startup hook has
472    //       pre-managed clients for us. In that case we want to avoid stomping on
473    //       anything that they have set up.
474    #[tracing::instrument(level = "info", skip(state, self))]
475    fn manage_existing_clients(&mut self, state: &mut State<Self>) -> Result<()> {
476        // We're not guaranteed that workspace indices are _always_ continuous from 0..n
477        // so we explicitly map tags to indices instead.
478        // We also exclude hidden workspaces as those can contain windows which are
479        // externally managed by a user written extension, which can lead to malformed
480        // internal state for those extensions when they restart.
481        let ws_map: HashMap<usize, String> = state
482            .client_set
483            .non_hidden_workspaces()
484            .map(|w| (w.id, w.tag.clone()))
485            .collect();
486
487        let first_tag = state.client_set.ordered_tags()[0].clone();
488
489        for id in self.existing_clients()? {
490            if !state.client_set.contains(&id) && existing_client_should_be_managed(self, id) {
491                let workspace_id = match self.get_prop(id, Atom::NetWmDesktop.as_ref()) {
492                    Ok(Some(Prop::Cardinal(ids))) => ids[0] as usize,
493                    _ => 0, // we know that we always have at least one workspace
494                };
495
496                let tag = ws_map.get(&workspace_id).unwrap_or(&first_tag);
497                let title = self.client_title(id)?;
498                info!(%id, %title, %tag, "attempting to manage existing client");
499                manage_without_refresh(id, Some(tag), state, self)?;
500            }
501        }
502
503        // If EWMH is enabled then we should have this property set to tell us what the previously
504        // active client was. If that client is not in the client set or the property is not set we
505        // default to forcing focus to the first available tag and whatever active client we have there
506        // as that is where we will have placed all existing clients.
507        match self.get_prop(state.root, Atom::NetActiveWindow.as_ref()) {
508            Ok(Some(Prop::Window(ids))) if state.client_set.contains(&ids[0]) => {
509                let id = ids[0];
510                info!(%id, "focusing _NET_ACTIVE_WINDOW client");
511                state.client_set.focus_client(&id);
512            }
513            _ => {
514                info!(%first_tag, "unable to determine an active window: focusing first tag");
515                state.client_set.focus_tag(&first_tag);
516            }
517        };
518
519        info!("triggering refresh");
520        self.refresh(state)
521    }
