33/// A window type to be specified when creating a new window in the X server

78/// A handle on a running X11 connection that we can use for issuing X requests.
79///
80/// XConn is intended as an abstraction layer to allow for communication with the underlying
81/// display system (assumed to be X) using whatever mechanism the implementer wishes. In theory, it
82/// should be possible to write an implementation that allows penrose to run on systems not using X
83/// as the windowing system but X idioms and high level event types / client interations are
84/// assumed.
85pub trait XConn {
86    /// The ID of the window manager root window.
87    fn root(&self) -> Xid;
88    /// Ask the X server for the dimensions of each currently available screen.
89    fn screen_details(&self) -> Result<Vec<Rect>>;
90    /// Ask the X server for the current (x, y) coordinate of the mouse cursor.
91    fn cursor_position(&self) -> Result<Point>;
92
93    /// Grab the specified key and mouse states, intercepting them for processing within
94    /// the window manager itself.
95    fn grab(&self, key_codes: &[KeyCode], mouse_states: &[MouseState]) -> Result<()>;
96    /// Block and wait for the next event from the X server so it can be processed.
97    fn next_event(&self) -> Result<XEvent>;
98    /// Flush any pending events to the X server.
99    fn flush(&self);
100
101    /// Look up the [Xid] of a given [Atom] name. If it is not currently interned, intern it.
102    fn intern_atom(&self, atom: &str) -> Result<Xid>;
103    /// Look up the string name of a given [Atom] by its [Xid].
104    fn atom_name(&self, xid: Xid) -> Result<String>;
105
106    /// Look up the current dimensions and position of a given client window.
107    fn client_geometry(&self, client: Xid) -> Result<Rect>;
108    /// Ask the X server for the IDs of all currently known client windows
109    fn existing_clients(&self) -> Result<Vec<Xid>>;
110
111    /// Map the given client window to the screen with its current geometry, making it visible.
112    fn map(&self, client: Xid) -> Result<()>;
113    /// Unmap the given client window from the screen, hiding it.
114    fn unmap(&self, client: Xid) -> Result<()>;
115    /// Kill the given client window, closing it.
116    fn kill(&self, client: Xid) -> Result<()>;
117    /// Set X input focus to be held by the given client window.
118    fn focus(&self, client: Xid) -> Result<()>;
119
120    /// Look up a specific property on a given client window.
121    fn get_prop(&self, client: Xid, prop_name: &str) -> Result<Option<Prop>>;
122    /// List the known property names set for a given client.
123    fn list_props(&self, client: Xid) -> Result<Vec<String>>;
124    /// Get the current [WmState] for a given client window.
125    fn get_wm_state(&self, client: Xid) -> Result<Option<WmState>>;
126    /// Request the [WindowAttributes] for a given client window from the X server.
127    fn get_window_attributes(&self, client: Xid) -> Result<WindowAttributes>;
128
129    /// Set the current [WmState] for a given client window.
130    fn set_wm_state(&self, client: Xid, wm_state: WmState) -> Result<()>;
131    /// Set a specific property on a given client window.
132    fn set_prop(&self, client: Xid, name: &str, val: Prop) -> Result<()>;
133    /// Delete a property for a given client window.
134    fn delete_prop(&self, client: Xid, prop_name: &str) -> Result<()>;
135    /// Set one or more [ClientAttr] for a given client window.
136    fn set_client_attributes(&self, client: Xid, attrs: &[ClientAttr]) -> Result<()>;
137    /// Set the [ClientConfig] for a given client window.
138    fn set_client_config(&self, client: Xid, data: &[ClientConfig]) -> Result<()>;
139    /// Send a [ClientMessage] to a given client.
140    fn send_client_message(&self, msg: ClientMessage) -> Result<()>;
141
142    /// Reposition the mouse cursor to the given (x, y) coordinates within the specified window.
143    /// This method should not be called directly: use `warp_pointer_to_window` or `warp_pointer_to_screen`
144    /// instead.
145    fn warp_pointer(&self, id: Xid, x: i16, y: i16) -> Result<()>;
146}

148/// Extended functionality for [XConn] impls in order to run the window manager.
149pub trait XConnExt: XConn + Sized {
150    /// Kill the focused client if there is one
151    fn kill_focused(&self, state: &mut State<Self>) -> Result<()> {
152        if let Some(&id) = state.client_set.current_client() {
153            self.kill(id)?;
154        }
155
156        Ok(())
157    }
158    /// Establish the window manager state for the given client window and refresh the
159    /// current X state.
160    fn manage(&self, id: Xid, state: &mut State<Self>) -> Result<()> {
161        trace!(%id, "managing new client");
162        manage_without_refresh(id, None, state, self)?;
163        self.refresh(state)
164    }
165
166    /// Remove the window manager state for the given client window and refresh the
167    /// current X state.
168    fn unmanage(&self, client: Xid, state: &mut State<Self>) -> Result<()> {
169        trace!(?client, "removing client");
170        self.modify_and_refresh(state, |cs| {
171            cs.remove_client(&client);
172        })
173    }
174
175    /// Display a client on the screen by mapping it and setting its WmState to Normal
176    /// This is idempotent if the client is already visible.
177    fn reveal(&self, client: Xid, cs: &ClientSet, mapped: &mut HashSet<Xid>) -> Result<()> {
178        self.set_wm_state(client, WmState::Normal)?;
179        self.map(client)?;
180        if cs.contains(&client) {
181            mapped.insert(client);
182        }
183
184        Ok(())
185    }
186
187    /// Hide a client by unmapping it and setting its WmState to Iconic
188    fn hide(
189        &self,
190        client: Xid,
191        mapped: &mut HashSet<Xid>,
192        pending_unmap: &mut HashMap<Xid, usize>,
193    ) -> Result<()> {
194        if !mapped.contains(&client) {
195            return Ok(());
196        }
197
198        self.set_client_attributes(client, &[ClientAttr::ClientUnmapMask])?;
199        self.unmap(client)?;
200        self.set_client_attributes(client, &[ClientAttr::ClientEventMask])?;
201        self.set_wm_state(client, WmState::Iconic)?;
202
203        mapped.remove(&client);
204        pending_unmap
205            .entry(client)
206            .and_modify(|count| *count += 1)
207            .or_insert(1);
208
209        Ok(())
210    }
211
212    /// Apply a pure function that modifies a [ClientSet] and then handle refreshing the
213    /// WindowManager state and associated X11 calls.
214    ///
215    /// This is the main logic that drives what the user will see on the screen in terms
216    /// of window placement, focus and borders. Everything is driven from a diff of the
217    /// pure ClientSet state before and after some mutating operation that was carried out
218    /// by `f`.
219    fn modify_and_refresh<F>(&self, state: &mut State<Self>, mut f: F) -> Result<()>
220    where
221        F: FnMut(&mut ClientSet),
222    {
223        f(&mut state.client_set); // mutating the existing state
224
225        let ss = state.position_and_snapshot(self);
226        state.diff.update(ss);
227
228        notify_killed(self, state)?;
229        set_window_props(self, state)?;
230        notify_hidden_workspaces(state);
231        self.position_clients(state)?;
232        set_window_visibility(self, state)?;
233        set_focus(self, state)?;
234        handle_pointer_change(self, state)?;
235
236        // TODO: clear enterWindow events from the event queue if this was because of mouse focus (?)
237
238        let mut hook = state.config.refresh_hook.take();
239        if let Some(ref mut h) = hook {
240            trace!("running user refresh hook");
241            if let Err(e) = h.call(state, self) {
242                error!(%e, "error returned from user refresh hook");
243            }
244        }
245        state.config.refresh_hook = hook;
246
247        Ok(())
248    }
249
250    /// Refresh the current X server state based on a diff of the current state against the state
251    /// when we last refreshed.
252    fn refresh(&self, state: &mut State<Self>) -> Result<()> {
253        self.modify_and_refresh(state, |_| ())
254    }
255
256    /// Check whether or not the given client should be assigned floating status or not.
257    fn client_should_float(&self, client: Xid, floating_classes: &[String]) -> Result<bool> {
258        trace!(%client, "fetching WmClass prop");
259        if let Some(Prop::UTF8String(strs)) = self.get_prop(client, Atom::WmClass.as_ref())? {
260            if strs.iter().any(|c| floating_classes.contains(c)) {
261                debug!(%client, ?floating_classes, "window has a floating class: setting to floating state");
262                return Ok(true);
263            }
264        }
265
266        trace!(%client, "fetching NetWmWindowType prop");
267        let window_types = self.get_prop(client, Atom::NetWmWindowType.as_ref())?;
268        debug!(?window_types, "client window types");
269
270        let float_types: Vec<&str> = AUTO_FLOAT_WINDOW_TYPES.iter().map(|a| a.as_ref()).collect();
271
272        let should_float = match window_types {
273            Some(Prop::Atom(atoms)) => atoms.iter().any(|a| float_types.contains(&a.as_ref())),
274            _ => false,
275        };
276
277        Ok(should_float)
278    }

280    /// Update the border color of the given client window.
281    fn set_client_border_color<C>(&self, id: Xid, color: C) -> Result<()>
282    where
283        C: Into<Color>,
284    {
285        let color = color.into();
286        self.set_client_attributes(id, &[ClientAttr::BorderColor(color.argb_u32())])
287    }
288
289    /// Set the initial window properties for a newly managed window.
290    fn set_initial_properties(&self, client: Xid, config: &Config<Self>) -> Result<()> {
291        let Config {
292            normal_border,
293            border_width,
294            ..
295        } = config;
296
297        let conf = &[ClientConfig::BorderPx(*border_width)];
298        let attrs = &[
299            ClientAttr::ClientEventMask,
300            ClientAttr::BorderColor(normal_border.argb_u32()),
301        ];
302
303        self.set_wm_state(client, WmState::Iconic)?;
304        self.set_client_attributes(client, attrs)?;
305        self.set_client_config(client, conf)
306    }
307
308    /// Update the geometry of a given client based on the given [Rect].
309    fn position_client(&self, client: Xid, mut r: Rect) -> Result<()> {
310        let p = Atom::WmNormalHints.as_ref();
311        if let Ok(Some(Prop::WmNormalHints(hints))) = self.get_prop(client, p) {
312            trace!(%client, ?hints, "client has WmNormalHints: applying size hints");
313            r = hints.apply_to(r);
314        }
315
316        trace!(%client, ?r, "positioning client");
317        self.set_client_config(client, &[ClientConfig::Position(r)])
318    }
319
320    /// Restack and set the geometry for an ordered list of client windows and their
321    /// associated positions. The provided positions are shrunk by the current border
322    /// size in order to position the windows correctly within the frame given by the
323    /// border.
324    ///
325    /// See `restack` for details of stacking order is determined.
326    fn position_clients(&self, state: &State<Self>) -> Result<()> {
327        let border = state.config.border_width;
328        let positions = &state.diff.after.positions;
329        let screen_positions: Vec<_> = state.client_set.screens().map(|s| s.r).collect();
330
331        self.restack(positions.iter().map(|(id, _)| id))?;
332
333        for &(c, mut r) in positions.iter() {
334            if !screen_positions.contains(&r) {
335                r = r.shrink_in(border);
336            }
337            self.position_client(c, r)?;
338        }
339
340        Ok(())
341    }
342
343    /// Restack the given windows in, each one above the last.
344    fn restack<'a, I>(&self, mut ids: I) -> Result<()>
345    where
346        I: Iterator<Item = &'a Xid>,
347    {
348        let mut previous = match ids.next() {
349            Some(id) => *id,
350            None => return Ok(()), // nothing to stack
351        };
352
353        for &id in ids {
354            self.set_client_config(id, &[ClientConfig::StackAbove(previous)])?;
355            previous = id;
356        }
357
358        Ok(())
359    }

361    /// Update the currently focused client and refresh the X state.
362    fn set_active_client(&self, client: Xid, state: &mut State<Self>) -> Result<()> {
363        self.modify_and_refresh(state, |cs| cs.focus_client(&client))
364    }

366    /// Warp the mouse cursor to the center of the given client window.
367    fn warp_pointer_to_window(&self, id: Xid) -> Result<()> {
368        let r = self.client_geometry(id)?;
369
370        self.warp_pointer(id, r.w as i16 / 2, r.h as i16 / 2)
371    }
372
373    /// Warp the mouse cursor to the center of the given screen.
374    fn warp_pointer_to_screen(&self, state: &mut State<Self>, screen_index: usize) -> Result<()> {
375        let maybe_screen = state.client_set.screens().find(|s| s.index == screen_index);
376
377        let screen = match maybe_screen {
378            Some(s) => s,
379            None => return Ok(()), // Unknown screen
380        };
381
382        if let Some(id) = screen.workspace.focus() {
383            return self.warp_pointer_to_window(*id);
384        }
385
386        let x = (screen.r.x + screen.r.w as i32 / 2) as i16;
387        let y = (screen.r.y + screen.r.h as i32 / 2) as i16;
388
389        self.warp_pointer(self.root(), x, y)
390    }

392    /// Fetch the value of all known properties for a given client window
393    fn all_props_for(&self, id: Xid) -> Result<HashMap<String, Prop>> {
394        self.list_props(id)?
395            .into_iter()
396            .map(|s| {
397                self.get_prop(id, &s)
398                    .map(|opt| (s, opt.expect("prop to be set")))
399            })
400            .collect()
401    }

403    /// Request the title of a given client window following ICCCM/EWMH standards.
404    fn window_title(&self, id: Xid) -> Result<String> {
405        match query::str_prop(Atom::WmName, id, self) {
406            Ok(Some(mut strs)) => Ok(strs.remove(0)),
407            _ => match query::str_prop(Atom::NetWmName, id, self)? {
408                Some(mut strs) => Ok(strs.remove(0)),
409                None => Ok("".to_owned()),
410            },
411        }
412    }

414    /// Check to see if a given client window supports a particular protocol or not
415    fn client_supports_protocol(&self, id: Xid, proto: &str) -> Result<bool> {
416        if let Some(Prop::Atom(protocols)) = self.get_prop(id, Atom::WmProtocols.as_ref())? {
417            Ok(protocols.iter().any(|p| p == proto))
418        } else {
419            Ok(false)
420        }
421    }

424    ///
425    /// **NOTE**: Not all programs set this property.

427        if let Ok(Some(Prop::Cardinal(vals))) = self.get_prop(id, "_NET_WM_PID") {

449// The main logic for inserting a new client into the StackSet without any refresh
450// of the X state. In normal window manager operation, the `manage` method on XConnExt
451// is always used: this is provided independently to support managing existing clients
452// on startup.
453pub(crate) fn manage_without_refresh<X: XConn>(
454    id: Xid,
455    tag: Option<&str>,
456    state: &mut State<X>,
457    x: &X,
458) -> Result<()> {
459    trace!(%id, "fetching WmTransientFor prop");
460    let (owned_tag, transient_for) = match x.get_prop(id, Atom::WmTransientFor.as_ref())? {
461        Some(Prop::Window(ids)) => {
462            let parent = ids[0];
463            let owned_tag = state
464                .client_set
465                .tag_for_client(&parent)
466                .or(tag)
467                .map(|t| t.to_string());
468
469            debug!(%id, %parent, ?owned_tag, "client is transient");
470            (owned_tag, Some(parent))
471        }
472
473        _ => (tag.map(|t| t.to_string()), None),
474    };
475
476    let should_float =
477        transient_for.is_some() || x.client_should_float(id, &state.config.floating_classes)?;
478
479    match owned_tag {
480        Some(tag) => state.client_set.insert_as_focus_for(tag.as_ref(), id),
481        None => state.client_set.insert(id),
482    }
483
484    if should_float {
485        debug!(%id, "client should float");
486        let r = floating_client_position(id, transient_for, state, x)?;
487        if state.client_set.float(id, r).is_err() {
488            error!(%id, "attempted to float client which was not in state");
489        }
490    }
491
492    let mut hook = state.config.manage_hook.take();
493    if let Some(ref mut h) = hook {
494        trace!("running user manage hook");
495        if let Err(e) = h.call(id, state, x) {
496            error!(%e, "error returned from user manage hook");
497        }
498    }
499    state.config.manage_hook = hook;
500
501    debug!(
502        floating=?state.client_set.floating, "floating clients"
503    );
504
505    Ok(())
506}

508/// When positioning a floating client we try to position them in priority order of:
509///   - the client's requested position if it is not at the origin
510///   - centered in their parent's screen (if transient)
511///   - centered in the focused screen
512fn floating_client_position<X: XConn>(
513    id: Xid,
514    transient_for: Option<Xid>,
515    state: &State<X>,
516    x: &X,
517) -> Result<Rect> {
518    trace!(%id, "fetching client geometry");
519    let r_initial = x.client_geometry(id)?;
520    debug!(?r_initial, "initial geometry");
521
522    if (r_initial.x, r_initial.y) != (0, 0) {
523        debug!(?r_initial, "accepting client's requested position");
524        return Ok(r_initial);
525    }
526
527    let r_parent = transient_for
528        .and_then(|parent| state.client_set.screen_for_client(&parent))
529        .unwrap_or(&state.client_set.screens.focus)
530        .r;
531    debug!(?r_parent, "parent geometry");
532
533    let r_final = r_initial.centered_in(&r_parent).unwrap_or_else(|| {
534        r_initial
535            .centered_in(&state.client_set.screens.focus.r)
536            .unwrap_or(r_initial)
537    });
538    debug!(?r_final, "final geometry");
539
540    Ok(r_final)
541}

543fn notify_killed<X: XConn>(x: &X, state: &mut State<X>) -> Result<()> {
544    for &c in state.diff.killed_clients() {
545        x.kill(c)?;
546    }

551fn set_window_props<X: XConn>(x: &X, state: &mut State<X>) -> Result<()> {
552    for &c in state.diff.new_clients() {
553        x.set_initial_properties(c, &state.config)?;
554    }
555
556    if let Some(focused) = state.diff.before.focused_client {
557        x.set_client_border_color(focused, state.config.normal_border)?;
558    }
559
560    if let Some(&focused) = state.client_set.current_client() {
561        trace!(?focused, "setting border for focused client");
562        x.set_client_border_color(focused, state.config.focused_border)?;
563    }
564
565    Ok(())
566}

568fn notify_hidden_workspaces<X: XConn>(state: &mut State<X>) {
569    let previous_visible_tags = state.diff.previous_visible_tags();
570
571    state
572        .client_set
573        .hidden_workspaces_mut()
574        .filter(|w| previous_visible_tags.contains(&w.tag.as_ref()))
575        .for_each(|ws| ws.broadcast_message(Hide));
576}

578// Warp the cursor if this diff resulted in a focus change
579fn handle_pointer_change<X: XConn>(x: &X, state: &mut State<X>) -> Result<()> {
580    if !state.config.focus_follow_mouse {
581        return Ok(());
582    }
583
584    trace!("checking if focus should change");
585    if !matches!(state.current_event, Some(XEvent::Enter(_))) {
586        if let Some(id) = state.diff.focused_client() {
587            trace!("focused client changed");
588            // NOTE: Some of the behaviour here is based on looking at whether or
589            //       not the focused client has changed position as part of this
590            //       diff. That is going to cause issues if and when mouse based
591            //       window movement is implemented.
592            let focus_changed = state.diff.focused_client_changed();
593            let focused_client_moved = state.diff.client_changed_position(&id);
594
595            if focus_changed || focused_client_moved {
596                trace!(
597                    focus_changed,
598                    focused_client_moved,
599                    "warping to focused client"
600                );
601                x.warp_pointer_to_window(id)?;
602            }
603        } else if let Some(index) = state.diff.newly_focused_screen() {
604            trace!(index, "screen changed: warping to screen");
605            x.warp_pointer_to_screen(state, index)?;
606        }
607    }
608
609    Ok(())
610}

612fn set_window_visibility<X: XConn>(x: &X, state: &mut State<X>) -> Result<()> {
613    for &c in state.diff.visible_clients() {
614        trace!(?c, "revealing client");
615        x.reveal(c, &state.client_set, &mut state.mapped)?;
616    }
617
618    for &c in state.diff.hidden_clients() {
619        trace!(?c, "hiding client");
620        x.hide(c, &mut state.mapped, &mut state.pending_unmap)?;
621    }
622
623    for &c in state.diff.withdrawn_clients() {
624        trace!(?c, "setting withdrawn state for client");
625        x.set_wm_state(c, WmState::Withdrawn)?;
626    }
627
628    Ok(())
629}
630
631fn set_focus<X: XConn>(x: &X, state: &mut State<X>) -> Result<()> {
632    if let Some(&id) = state.client_set.current_client() {
633        x.focus(id)
634    } else {
635        x.focus(state.root)
636    }
637}
