12/// Wrapper around the low level X event types that correspond to request / response data when
13/// communicating with the X server itself.
14///
15/// The variant names and data have developed with the reference xcb implementation in mind but
16/// should be applicable for all back ends.
17#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
18#[derive(Debug, Clone, PartialEq, Eq, Hash)]
19#[non_exhaustive]
20pub enum XEvent {
21    /// A message has been sent to a particular client
22    ClientMessage(ClientMessage),
23    /// Client config has changed in some way
24    ConfigureNotify(ConfigureEvent),
25    /// A client is requesting to be repositioned
26    ConfigureRequest(ConfigureEvent),
27    /// The mouse pointer has entered a new client window
28    Enter(PointerChange),
29    /// A part or all of a client has become visible
30    Expose(ExposeEvent),
31    /// A client should have focus
32    FocusIn(Xid),
33    /// A client window has been closed
34    Destroy(Xid),
35    /// A grabbed key combination has been entered by the user
36    KeyPress(KeyCode),
37    /// The mouse pointer has left the current client window
38    Leave(PointerChange),
39    /// Keybindings have changed
40    MappingNotify,
41    /// A client window is requesting to be positioned and rendered on the screen.
42    MapRequest(Xid),
43    /// A mouse button has been pressed or released
44    MouseEvent(MouseEvent),
45    /// The mouse has moved while a grabbed mouse state is held
46    MotionNotify(MotionNotifyEvent),
47    /// A client property has changed in some way
48    PropertyNotify(PropertyEvent),
49    /// A randr action has occured (new outputs, resolution change etc)
50    RandrNotify,
51    /// A window is requesting that it be resized
52    ResizeRequest(ResizeRequestEvent),
53    /// Focus has moved to a different screen
54    ScreenChange,
55    /// A client is being unmapped
56    UnmapNotify(Xid),
57}

86/// Known common client message formats.
87#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
88pub enum ClientMessageKind {
89    /// Inform a client that it is being closed
90    DeleteWindow(Xid),
91    /// Request that a client take input focus
92    TakeFocus(Xid),
93    /// Take ownership of the systray
94    ///
95    /// Args are the id of the root window and id of the window being used as a systray
96    TakeSystrayOwnership(Xid, Xid),
97    /// Inform an embedded window that it has gained focus
98    XEmbedFocusIn(Xid, Xid),
99    /// Inform an embedded window that it has been blocked by a modal dialog
100    XEmbedModalityOn(Xid, Xid),
101    /// Inform a window that it is being embedded
102    XEmbedNotify(Xid, Xid),
103    /// Inform an embedded window that it is now active
104    XEmbedWindowActivate(Xid, Xid),
105}
106
107impl ClientMessageKind {
108    /// Build a default [ClientMessage] compatible with X11 / XCB formats.
109    ///
110    /// Most impls of `X*` traits should be able to use the default data generated by this method,
111    /// but if you need to send something else, you can always construct the `ClientMessage`
112    /// explicitly.
113    pub fn as_message<X>(&self, q: &X) -> Result<ClientMessage>
114    where
115        X: XConn,
116    {
117        let proto_msg = |id: Xid, atom: Atom| {
118            let proto = Atom::WmProtocols.as_ref();
119            let data = &[*q.intern_atom(atom.as_ref())?, 0, 0, 0, 0];
120            let mask = ClientEventMask::NoEventMask;
121
122            Ok(ClientMessage::new(id, mask, proto, data.into()))
123        };
124
125        // https://specifications.freedesktop.org/xembed-spec/xembed-spec-latest.html
126        let xembed_version = 0;
127        let notify = 0;
128        let activate = 1;
129        let focus_in = 4;
130        let modality_on = 10;
131
132        let xembed_msg = |id: Xid, embedder: Xid, kind: u32| {
133            let atom = Atom::XEmbed.as_ref();
134            let data = &[0, kind, 0, *embedder, xembed_version];
135            let mask = ClientEventMask::SubstructureNotify;
136
137            Ok(ClientMessage::new(id, mask, atom, data.into()))
138        };
139
140        match self {
141            ClientMessageKind::DeleteWindow(id) => proto_msg(*id, Atom::WmDeleteWindow),
142            ClientMessageKind::TakeFocus(id) => proto_msg(*id, Atom::WmTakeFocus),
143
144            ClientMessageKind::TakeSystrayOwnership(root_id, systray_id) => {
145                let atom = Atom::Manager.as_ref();
146                let systray = q.intern_atom(Atom::NetSystemTrayS0.as_ref())?;
147                let data = &[0, *systray, **systray_id, 0, 0];
148                let mask = ClientEventMask::SubstructureNotify;
149
150                Ok(ClientMessage::new(*root_id, mask, atom, data.into()))
151            }
152
153            ClientMessageKind::XEmbedFocusIn(id, other) => xembed_msg(*id, *other, focus_in),
154            ClientMessageKind::XEmbedModalityOn(id, other) => xembed_msg(*id, *other, modality_on),
155            ClientMessageKind::XEmbedNotify(id, other) => xembed_msg(*id, *other, notify),
156            ClientMessageKind::XEmbedWindowActivate(id, other) => xembed_msg(*id, *other, activate),
157        }
158    }
159}

173/// The raw data contained in a [`ClientMessage`]
174#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
175#[derive(Debug, Clone, PartialEq, Eq, Hash)]
176pub enum ClientMessageData {
177    /// Slice of u8
178    U8([u8; 20]),
179    /// Slice of u16
180    U16([u16; 10]),
181    /// Slice of u32
182    U32([u32; 5]),
183}
184
185macro_rules! cast_slice {
186    ($s:expr, $t:ty) => {
187        $s.iter().map(|&v| v as $t).collect::<Vec<$t>>()
188    };
189}
190
191impl ClientMessageData {
192    /// Convert this client message into a single data format
193    ///
194    /// The number of raw values will be maintained but this allows you to have a consistant
195    /// interface without needing to match on the variant or cast all the time.
196    pub fn as_usize(&self) -> Vec<usize> {
197        match self {
198            Self::U8(data) => cast_slice!(data, usize),
199            Self::U16(data) => cast_slice!(data, usize),
200            Self::U32(data) => cast_slice!(data, usize),
201        }
202    }
203}
204
205macro_rules! __impl_client_message_data(
206    { $t:ty; $count:expr, $variant:expr, $method:ident } => {
207        impl ClientMessageData {
208            /// Convert this client message into a single data format
209            ///
210            /// The number of raw values will be maintained but this allows you to have a consistant
211            /// interface without needing to match on the variant or cast all the time.
212            pub fn $method(&self) -> Vec<$t> {
213                match self {
214                    Self::U8(data) => cast_slice!(data, $t),
215                    Self::U16(data) => cast_slice!(data, $t),
216                    Self::U32(data) => cast_slice!(data, $t),
217                }
218            }
219        }
220        impl From<[$t; $count]> for ClientMessageData {
221            fn from(data: [$t; $count]) -> Self {
222                $variant(data)
223            }
224        }
225        impl From<&[$t; $count]> for ClientMessageData {
226            fn from(data: &[$t; $count]) -> Self {
227                $variant(*data)
228            }
229        }
230        impl TryFrom<&[$t]> for ClientMessageData {
231            type Error = std::array::TryFromSliceError;
232
233            fn try_from(data: &[$t]) -> std::result::Result<Self, Self::Error> {
234                Ok($variant(<[$t; $count]>::try_from(data)?))
235            }
236        }
237    }
238);
239
240__impl_client_message_data!(u8; 20, ClientMessageData::U8, as_u8);
241__impl_client_message_data!(u16; 10, ClientMessageData::U16, as_u16);
242__impl_client_message_data!(u32; 5, ClientMessageData::U32, as_u32);

244/// A client message that needs to be parsed and handled based on its type
245#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
246#[derive(Debug, Clone, PartialEq, Eq, Hash)]
247pub struct ClientMessage {
248    /// The ID of the window that sent the message
249    pub id: Xid,
250    /// The mask to use when sending the event
251    pub mask: ClientEventMask,
252    /// The data type being set
253    pub dtype: String,
254    /// The raw data being sent in this message
255    pub data: ClientMessageData,
256}
257
258impl ClientMessage {
259    /// Try to build a new ClientMessage. Fails if the data is invalid
260    pub fn new(
261        id: Xid,
262        mask: ClientEventMask,
263        dtype: impl Into<String>,
264        data: ClientMessageData,
265    ) -> Self {
266        Self {
267            id,
268            mask,
269            dtype: dtype.into(),
270            data,
271        }
272    }
273}
