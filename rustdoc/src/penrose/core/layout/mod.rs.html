25/// A [Layout] is responsible for positioning a [Stack] of clients in a given coordinate space denoting
26/// the dimensions of users display.
27///
28/// Mutating the state of a Layout is possible by sending it a [Message] which can then either modify
29/// the existing layout (e.g. increase the number of clients positioned in a "main" area) or replace
30/// the existing Layout with a new one. There is no requirement to be able to handle all message types.
31pub trait Layout: Send + Sync {
32    /// A short display name for this Layout, appropriate for rendering in a status bar as an indicator
33    /// of which layout is currently being used.
34    fn name(&self) -> String;
35
36    /// Provide a clone of this [Layout] wrapped as a trait object. (Trait objects can not require
37    /// Clone directly)
38    fn boxed_clone(&self) -> Box<dyn Layout>;
39
40    /// Convert to a trait object.
41    fn boxed(self) -> Box<dyn Layout>
42    where
43        Self: Sized + 'static,
44    {
45        Box::new(self)
46    }
47
48    /// Generate screen positions for clients on a given [crate::pure::Workspace].
49    ///
50    /// If you do not need to know the details of the workspace being laid out, you should can use the
51    /// default implementation of this methods which calls [Layout::layout] if there are any clients
52    /// present and [Layout::layout_empty] if not.
53    ///
54    /// # Positioning clients
55    /// For each client that should be shown on the screen a pair of its [Xid] and a [Rect] should be
56    /// provided, indicating the screen position the client should be placed at. To hide a client that
57    /// was present in the [crate::pure::Workspace] simply do not provide a position for it. (You may also provide
58    /// positions for clients that were not present in the input if you have the [Xid] available.)
59    ///
60    /// The order in which the ([Xid], [Rect]) pairs are returned determines the stacking order on the
61    /// screen. It does not have to match the stack order of the clients within the [crate::pure::Workspace].
62    ///
63    /// # Returning a new layout
64    /// When a layout is run it may optionally replace itself with a new [Layout]. If `Some(layout)`
65    /// is returned from this method, it will be swapped out for the current one after the provided
66    /// positions have been applied.
67    #[allow(clippy::type_complexity, unused_variables)]
68    fn layout_workspace(
69        &mut self,
70        tag: &str,
71        stack: &Option<Stack<Xid>>,
72        r: Rect,
73    ) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>) {
74        match stack {
75            Some(s) => self.layout(s, r),
76            None => self.layout_empty(r),
77        }
78    }
79
80    /// Generate screen positions for clients from a given [Stack].
81    ///
82    /// See [Layout::layout_workspace] for details of how positions should be returned.
83    #[allow(clippy::type_complexity)]
84    fn layout(&mut self, s: &Stack<Xid>, r: Rect) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>);
85
86    /// Generate screen positions for an empty [Stack].
87    ///
88    /// See [Layout::layout_workspace] for details of how positions should be returned.
89    #[allow(clippy::type_complexity, unused_variables)]
90    fn layout_empty(&mut self, r: Rect) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>) {
91        (None, vec![])
92    }
93
94    /// Process a dynamic [Message].
95    ///
96    /// See the trait level docs for details on what is possible with messages.
97    fn handle_message(&mut self, m: &Message) -> Option<Box<dyn Layout>>;
98}

101    fn clone(&self) -> Self {
102        self.boxed_clone()
103    }
104}

120/// A stack of [Layout] options for use on a particular [crate::pure::Workspace].
121///
122/// The [Stack] itself acts as a [Layout], deferring all operations to the
123/// currently focused Layout.
124pub type LayoutStack = Stack<Box<dyn Layout>>;
125
126impl Default for LayoutStack {
127    fn default() -> Self {
128        stack!(Box::<MainAndStack>::default())
129    }
130}
131
132impl LayoutStack {
133    /// Run the currently focused [Layout] and return the positions it generates.
134    ///
135    /// If the layout being run wants to be replaced with a new layout, swap it
136    /// out for the new one in its current position in the [Stack].
137    pub fn run_and_replace<F>(&mut self, f: F) -> Vec<(Xid, Rect)>
138    where
139        F: FnOnce(&mut Box<dyn Layout>) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>),
140    {
141        let (new_focus, rs) = (f)(&mut self.focus);
142
143        if let Some(mut new) = new_focus {
144            self.swap_focus(&mut new);
145        }
146
147        rs
148    }
149
150    /// Send the given [Message] to the currently active [Layout].
151    pub fn handle_message<M>(&mut self, m: M)
152    where
153        M: IntoMessage,
154    {
155        let m = m.into_message();
156
157        if let Some(mut new) = self.focus.handle_message(&m) {
158            swap(&mut self.focus, &mut new);
159        }
160    }
161
162    /// Send the given [Message] to every [Layout] in this stack rather that just the
163    /// currently active one.
164    pub fn broadcast_message<M>(&mut self, m: M)
165    where
166        M: IntoMessage,
167    {
168        let m = m.into_message();
169
170        for l in self.iter_mut() {
171            if let Some(mut new) = l.handle_message(&m) {
172                swap(l, &mut new);
173            }
174        }
175    }
176}
177
178impl Layout for LayoutStack {
179    fn name(&self) -> String {
180        self.focus.name()
181    }
182
183    fn boxed_clone(&self) -> Box<dyn Layout> {
184        Box::new(self.clone())
185    }
186
187    fn layout_workspace(
188        &mut self,
189        tag: &str,
190        stack: &Option<Stack<Xid>>,
191        r: Rect,
192    ) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>) {
193        (
194            None,
195            self.run_and_replace(|l| l.layout_workspace(tag, stack, r)),
196        )
197    }
198
199    fn layout(&mut self, s: &Stack<Xid>, r: Rect) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>) {
200        (None, self.run_and_replace(|l| l.layout(s, r)))
201    }
202
203    fn layout_empty(&mut self, r: Rect) -> (Option<Box<dyn Layout>>, Vec<(Xid, Rect)>) {
204        (None, self.run_and_replace(|l| l.layout_empty(r)))
205    }
206
207    fn handle_message(&mut self, m: &Message) -> Option<Box<dyn Layout>> {
208        let new_focus = self.focus.handle_message(m);
209
210        if let Some(mut new) = new_focus {
211            self.swap_focus(&mut new);
212        }
213
214        None
215    }
216}
