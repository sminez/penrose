49impl<C> State<C>
50where
51    C: Conn,
52{
53    pub(crate) fn try_new(config: Config<C>, conn: &mut C) -> Result<Self> {
54        let mut client_set = StackSet::try_new(
55            config.default_layouts.clone(),
56            config.tags.iter(),
57            conn.screen_details()?,
58        )?;
59
60        let ss = client_set.snapshot(vec![]);
61        let diff = Diff::new(ss.clone(), ss);
62        let conn_state = conn.initial_state();
63
64        Ok(Self {
65            config,
66            client_set,
67            conn_state,
68            extensions: AnyMap::new(),
69            root: conn.root(),
70            current_event: None,
71            diff,
72            running: false,
73            held_mouse_state: None,
74        })
75    }
76
77    /// The WinId of the root window for the running [WindowManager].
78    pub fn root(&self) -> WinId {
79        self.root
80    }
81
82    /// The set of all client windows currently mapped to a screen.
83    pub fn mapped_clients(&self) -> HashSet<WinId> {
84        self.client_set.clients().cloned().collect()
85    }
86
87    /// The event currently being processed.
88    pub fn current_event(&self) -> Option<&C::Event> {
89        self.current_event.as_ref()
90    }
91
92    /// Get access to a shared state extension.
93    ///
94    /// To add an extension to [State] before starting the Window Manager, see the
95    /// [WindowManager::add_extension] method. To add an extension dynamically
96    /// when you have access to [State], see [State::add_extension].
97    ///
98    /// # Errors
99    /// Returns `Error::UnknownStateExtension` if there is no extension of type `E`.
100    pub fn extension<E: Any>(&self) -> Result<Arc<RefCell<E>>> {
101        self.extensions
102            .get()
103            .cloned()
104            .ok_or(Error::UnknownStateExtension {
105                type_id: TypeId::of::<E>(),
106            })
107    }
108
109    /// Get access to a shared state extension or set it using Default.
110    pub fn extension_or_default<E: Default + Any>(&mut self) -> Arc<RefCell<E>> {
111        if !self.extensions.contains::<Arc<RefCell<E>>>() {
112            self.add_extension(E::default());
113        }
114
115        self.extension().expect("to have defaulted if missing")
116    }
117
118    /// Remove a shared state extension entirely.
119    ///
120    /// Returns `None` if there is no extension of type `E` or if that extension
121    /// is currently being held by another thread.
122    pub fn remove_extension<E: Any>(&mut self) -> Option<E> {
123        let arc: Arc<RefCell<E>> = self.extensions.remove()?;
124
125        // If there is only one strong reference to this state then we'll be able to
126        // try_unwrap it and return the underlying `E`. If not the this fails so we
127        // need to store it back in the extensions anymap.
128        match Arc::try_unwrap(arc) {
129            Ok(rc) => Some(rc.into_inner()),
130            Err(arc) => {
131                self.extensions.insert(arc);
132                None
133            }
134        }
135    }
136
137    /// Add a typed [State] extension to this State.
138    pub fn add_extension<E: Any>(&mut self, extension: E) {
139        self.extensions.insert(Arc::new(RefCell::new(extension)));
140    }
141
142    pub(crate) fn position_and_snapshot(&mut self, conn: &mut C) -> Snapshot<WinId> {
143        let positions = self.visible_client_positions(conn);
144        self.client_set.snapshot(positions)
145    }
146
147    /// Run the per-workspace layouts to get a screen position for each visible client. Floating clients
148    /// are placed above stacked clients, clients per workspace are stacked in the order they are returned
149    /// from the layout.
150    pub(crate) fn visible_client_positions(&mut self, conn: &mut C) -> Vec<(WinId, Rect)> {
151        let mut float_positions: Vec<(WinId, Rect)> = Vec::new();
152        let mut positions: Vec<(WinId, Rect)> = Vec::new();
153
154        // pop the layout hook off of `state` so that we can pass state into it
155        let mut hook = self.config.layout_hook.take();
156
157        let scs: Vec<ScreenClients<WinId>> = self
158            .client_set
159            .screens
160            .iter()
161            .map(|s| s.screen_clients(&self.client_set.floating))
162            .collect();
163
164        for (i, sc) in scs.into_iter().enumerate() {
165            let ScreenClients {
166                floating,
167                tiling,
168                tag,
169                r_s,
170            } = sc;
171
172            // Sort out the floating client positions first
173            for (c, r_c) in floating.iter() {
174                float_positions.push((*c, r_c.applied_to(&r_s)));
175            }
176
177            // Next run layout functions for each workspace on a visible screen
178            let stack_positions = match hook {
179                Some(ref mut h) => {
180                    let r_s = h.transform_initial_for_screen(i, r_s, self, conn);
181                    let s = self.client_set.screens.iter_mut().nth(i).unwrap();
182                    let initial = s.workspace.apply_layout(&tag, &tiling, r_s);
183                    h.transform_positions_for_screen(i, r_s, initial, self, conn)
184                }
185                None => {
186                    let s = self.client_set.screens.iter_mut().nth(i).unwrap();
187                    s.workspace.apply_layout(&tag, &tiling, r_s)
188                }
189            };
190
191            positions.extend(stack_positions.into_iter().rev());
192        }
193
194        float_positions.reverse();
195        positions.extend(float_positions);
196
197        // Restore the layout hook
198        self.config.layout_hook = hook;
199
200        positions
201    }
202}

204/// The user specified config options for how the window manager should run
205pub struct Config<C>
206where
207    C: Conn,
208{
209    /// The RGBA color to use for normal (unfocused) window borders
210    pub normal_border: Color,
211    /// The RGBA color to use for the focused window border
212    pub focused_border: Color,
213    /// The width in pixels to use for drawing window borders
214    pub border_width: u32,
215    /// Whether or not the mouse entering a new window should set focus
216    pub focus_follow_mouse: bool,
217    /// The stack of layouts to use for each workspace
218    pub default_layouts: LayoutStack,
219    /// The ordered set of workspace tags to use on window manager startup
220    pub tags: Vec<String>,
221    /// Window classes that should always be assigned floating positions rather than tiled
222    pub floating_classes: Vec<String>,
223    /// A [StateHook] to run before entering the main event loop
224    pub startup_hook: Option<Box<dyn StateHook<C>>>,
225    /// A [StateHook] to run before processing each [XEvent]
226    pub event_hook: Option<Box<dyn EventHook<C>>>,
227    /// A [ManageHook] to run after each new window becomes managed by the window manager
228    pub manage_hook: Option<Box<dyn ManageHook<C>>>,
229    /// A [StateHook] to run every time the on screen X state is refreshed
230    pub refresh_hook: Option<Box<dyn StateHook<C>>>,
231    /// A [LayoutHook] to run when positioning clients on the screen
232    pub layout_hook: Option<Box<dyn LayoutHook<C>>>,
233}

351/// A top level struct holding all of the state required to run as an X11 window manager.
352///
353/// This allows for final configuration to be carried out before entering the main event
354/// loop.
355#[derive(Debug)]
356pub struct WindowManager<C>
357where
358    C: Conn,
359{
360    conn: C,
361    /// The mutable [State] of the window manager
362    pub state: State<C>,
363    key_bindings: KeyBindings<C>,
364    mouse_bindings: MouseBindings<C>,
365}
366
367impl<C> WindowManager<C>
368where
369    C: Conn,
370{
371    /// Construct a new [WindowManager] with the provided config and X connection.
372    ///
373    /// If you need to set [State] extensions, call [WindowManager::add_extension] after
374    /// constructing your initial WindowManager.
375    pub fn new(
376        config: Config<C>,
377        key_bindings: KeyBindings<C>,
378        mouse_bindings: MouseBindings<C>,
379        mut conn: C,
380    ) -> Result<Self> {
381        let state = State::try_new(config, &mut conn)?;
382
383        Ok(Self {
384            conn,
385            state,
386            key_bindings,
387            mouse_bindings,
388        })
389    }
390
391    /// Add a typed [State] extension to this WindowManager.
392    pub fn add_extension<E: Any>(&mut self, extension: E) {
393        self.state.add_extension(extension);
394    }
395
396    /// Start the WindowManager and run it until told to exit.
397    ///
398    /// Any provided startup hooks will be run after setting signal handlers and grabbing
399    /// key / mouse bindings from the X server. Any set up you need to do should be run
400    /// explicitly before calling this method or as part of a startup hook.
401    ///
402    /// ## Existing clients
403    /// An attempt will be made to pull any existing clients already present into the current
404    /// WindowManager state. This is done on a "best effort" basis to manage existing clients on
405    /// the workspaces they were present on previously. If you are planning on making use of this
406    /// functionality for more than recovering from a crash it is advised that you add EWMH hooks
407    /// to your Config so that there is more information available to correctly position your
408    /// existing clients.
409    /// Startup hooks are run before this takes place so that there is an opportunity to handle
410    /// restoring any state being held outside of the main WindowManager data structures.
411    ///
412    /// > **NOTE**: This is not guaranteed to preserve the stacking order or correctly handle any
413    /// > clients that were on invisible workspaces / workspaces that no longer exist and that the
414    /// > workspace containing the previously active client will be placed on the first available
415    /// > screen.
416    pub fn run(mut self) -> Result<()> {
417        info!("registering SIGCHILD signal handler");
418        // SAFETY: there is no previous signal handler so we are safe to set our own without needing
419        //         to worry about UB from the previous handler being invalid.
420        if let Err(e) = unsafe { signal(Signal::SIGCHLD, SigHandler::SigIgn) } {
421            panic!("unable to set signal handler: {}", e);
422        }
423
424        let key_codes: Vec<_> = self.key_bindings.keys().copied().collect();
425        let mouse_states: Vec<_> = self.mouse_bindings.keys().cloned().collect();
426        self.conn.grab(&key_codes, &mouse_states)?;
427
428        if let Some(mut h) = self.state.config.startup_hook.take() {
429            trace!("running user startup hook");
430            if let Err(e) = h.call(&mut self.state, &mut self.conn) {
431                error!(%e, "error returned from user startup hook");
432            }
433        }
434
435        info!("managing existing clients");
436        self.conn.manage_existing_clients(&mut self.state)?;
437        self.state.running = true;
438
439        debug!("entering main run loop");
440        while self.state.running {
441            match self.conn.next_event() {
442                Ok(event) => {
443                    let span = span!(target: "penrose", Level::INFO, "Event", %event);
444                    let _enter = span.enter();
445                    trace!(details = ?event, "event details");
446                    self.state.current_event = Some(event.clone());
447
448                    if let Err(e) = self.handle_event(event) {
449                        error!(%e, "Error handling Event");
450                    }
451                    self.conn.flush();
452                    self.state.current_event = None;
453                }
454
455                Err(e) => self.handle_error(e),
456            }
457        }
458
459        Ok(())
460    }
461
462    fn handle_event(&mut self, event: C::Event) -> Result<()> {
463        let WindowManager {
464            conn,
465            state,
466            key_bindings,
467            mouse_bindings,
468        } = self;
469
470        let mut hook = state.config.event_hook.take();
471        let should_run = match hook {
472            Some(ref mut h) => {
473                trace!("running user event hook");
474                match h.call(&event, state, conn) {
475                    Ok(should_run) => should_run,
476                    Err(e) => {
477                        error!(%e, "error returned from user event hook");
478                        true
479                    }
480                }
481            }
482
483            None => true,
484        };
485        state.config.event_hook = hook;
486
487        if !should_run {
488            trace!("User event hook returned false: skipping default handling");
489            return Ok(());
490        }
491
492        conn.handle_event(event, key_bindings, mouse_bindings, state)
493    }
494
495    fn handle_error(&mut self, e: Error) {
496        match e {
497            // If we get an error from the XConn telling us that a client ID is unknown then
498            // we need to make sure that we remove any reference to it from our internal state
499            Error::UnknownClient(id) => {
500                debug!(%id, "Conn encountered an error due to an unknown client ID: removing client");
501                self.state.client_set.remove_client(&id);
502            }
503
504            _ => error!(%e, "Unhandled error pulling next event"),
505        }
506    }
507}

514    fn stack_order(cs: &ClientSet) -> Vec<u32> {
515        let positions = cs.visible_client_positions();
516        positions.iter().map(|&(id, _)| *id).collect()
517    }

519    #[test]
520    fn floating_client_positions_are_respected() {
521        let mut s = test_xid_stack_set(5, 2);
522
523        for n in 0..4 {
524            s.insert(WinId(n));
525        }
526
527        let r = Rect::new(50, 50, 50, 50);
528        s.float_unchecked(WinId(1), r);

535    #[test]
536    fn floating_clients_stay_on_their_assigned_screen() {
537        let mut s = test_xid_stack_set(5, 2);
538
539        for n in 0..4 {
540            s.insert(WinId(n));
541        }
542
543        let r = Rect::new(50, 50, 50, 50);
544        s.float_unchecked(WinId(1), r);
545
546        let positions = s.visible_client_positions();
547
548        assert!(positions.contains(&(WinId(1), r)), "{positions:?}");
549
550        // If we move the client to tag 2 on the second screen then it should
551        // change position and be relative to that screen instead
552        s.move_client_to_tag(&WinId(1), "2");
553        let positions = s.visible_client_positions();
554
555        assert!(!positions.contains(&(WinId(1), r)), "{positions:?}");
556        assert!(
557            positions.contains(&(WinId(1), Rect::new(1050, 2050, 50, 50))),
558            "{positions:?}"
559        );
560    }
561
562    #[test]
563    fn floating_windows_are_returned_last() {
564        let mut s = test_xid_stack_set(5, 2);
565
566        for n in 1..6 {
567            s.insert(WinId(n));
568        }
569
570        s.float_unchecked(WinId(2), Rect::new(0, 0, 42, 42));
571        s.float_unchecked(WinId(3), Rect::new(0, 0, 69, 69));
572
573        assert_eq!(stack_order(&s), vec![1, 4, 5, 2, 3]);
574    }

576    #[test]
577    fn newly_added_windows_are_below_floating() {
578        let mut s = test_xid_stack_set(5, 2);
579
580        for n in 1..6 {
581            s.insert(WinId(n));
582        }
583
584        s.float_unchecked(WinId(2), Rect::new(0, 0, 42, 42));
585        s.float_unchecked(WinId(3), Rect::new(0, 0, 69, 69));
586
587        s.insert(WinId(6));
588
589        assert_eq!(stack_order(&s), vec![1, 4, 5, 6, 2, 3]);
590    }
591
592    #[test]
593    fn floating_clients_dont_break_insert_focus() {
594        let mut s = test_xid_stack_set(1, 1);
595
596        s.insert_at(Position::Focus, WinId(0));
597        s.float_unchecked(WinId(0), Rect::new(0, 0, 42, 42));
598
599        assert_eq!(s.current_client(), Some(&WinId(0)));
600
601        // Each time we add a client it should be the focus
602        // and the floating window should be stacked above
603        // all others.
604        let mut expected = vec![0];
605        for n in 1..=5 {
606            s.insert_at(Position::Focus, WinId(n));
607            assert_eq!(s.current_client(), Some(&WinId(n)));
608
609            // Tiled position ordering is reversed in visible_client_positions
610            // in order to ensure that when we restack, the order returned
611            // is from bottom -> top of the stack to make `restack` simpler to
612            // implement.
613            expected.insert(expected.len() - 1, n);
614            assert_eq!(stack_order(&s), expected, "{:?}", s.current_stack());
615        }
616    }
